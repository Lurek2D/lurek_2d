1//! Agent orchestration logic extracted from Lua runtime glue.
2//! Owns batch-task data contracts, callback ID packing, and system-context assembly.
3//! This module is runtime-agnostic and intentionally free of `mlua` types.
4
5use crate::agent::{AISystemState, AgentState};
6use std::collections::HashMap;
7
8/// One queued batch task derived from an agent configuration.
9#[derive(Clone)]
10pub struct AgentBatchTask {
11    /// Position index used to key the result in the batch output table.
12    pub agent_idx: usize,
13    /// Instruction sent to the LLM.
14    pub instruction: String,
15    /// Agent state snapshot for this task.
16    pub state: AgentState,
17    /// Optional pre-built system block that overrides the agent's own context assembly.
18    pub system_override: Option<String>,
19}
20
21/// Encodes `(batch_id, task_idx)` into a single callback ID.
22pub fn pack_batch_callback_id(batch_id: usize, task_idx: usize) -> usize {
23    (batch_id * 1000) + task_idx
24}
25
26/// Decodes a packed callback ID back into `(batch_id, task_idx)`.
27pub fn unpack_batch_callback_id(callback_id: usize) -> (usize, usize) {
28    (callback_id / 1000, callback_id % 1000)
29}
30
31/// Builds the system context block for the given prompt and optional target agent.
32pub fn build_system_context(
33    system_state: &AISystemState,
34    agents: &HashMap<String, AgentState>,
35    instruction: &str,
36    include_instructions: &[String],
37    agent_name: Option<&str>,
38) -> String {
39    let mut block = system_state.build_context(instruction, include_instructions);
40
41    if let Some(name) = agent_name {
42        if let Some(agent) = agents.get(name) {
43            if !agent.description.is_empty() {
44                block.push_str(&format!("\n\n[Agent: {}]:\n{}", name, agent.description));
45            }
46            let agent_system = agent.build_system_block();
47            if !agent_system.is_empty() {
48                block.push_str(&format!("\n\n{}", agent_system));
49            }
50        }
51    }
52
53    block
54}
55
56/// Builds a batch task for a named agent, injecting system context.
57pub fn make_system_task(
58    system_state: &AISystemState,
59    agents: &HashMap<String, AgentState>,
60    agent_name: &str,
61    task_idx: usize,
62    instruction: String,
63    include_instructions: &[String],
64) -> Result<AgentBatchTask, String> {
65    let agent = agents
66        .get(agent_name)
67        .ok_or_else(|| format!("agent '{}' not found", agent_name))?
68        .clone();
69
70    let system_block = build_system_context(
71        system_state,
72        agents,
73        &instruction,
74        include_instructions,
75        Some(agent_name),
76    );
77
78    Ok(AgentBatchTask {
79        agent_idx: task_idx,
80        instruction,
81        state: agent,
82        system_override: Some(system_block),
83    })
84}

1//! Manages CPU texture ingestion and staging before GPU-side renderer upload and sampling.
2//! Decodes files and raw buffers into validated RGBA payloads keyed in slot-map storage.
3//! Applies premultiplied-alpha conversion paths to align blending behavior with render expectations.
4//! Tracks texture color-space intent so pipelines can distinguish sRGB and linear content.
5//! Supplies safe construction and validation helpers used by asset loading and runtime creation flows.
6
7use crate::log_msg;
8use crate::render::renderer::TextureData;
9use crate::runtime::error::{EngineError, EngineResult};
10use crate::runtime::log_messages::TX01_TEX_DECODED;
11use crate::runtime::resource_keys::TextureKey;
12use slotmap::SlotMap;
13use std::path::Path;
14/// Texture color space stored alongside decoded pixels.
15#[derive(Clone, Copy, Debug, PartialEq, Eq)]
16pub enum TextureColorSpace {
17    /// Standard sRGB texture data.
18    Srgb,
19    /// Linear texture data.
20    Linear,
21}
22/// CPU-side texture handle and its uploaded dimensions.
23pub struct Texture {
24    /// Slot-map key for the backing texture data.
25    pub key: TextureKey,
26    /// Texture width in pixels.
27    pub width: u32,
28    /// Texture height in pixels.
29    pub height: u32,
30}
31/// Premultiply RGB channels by alpha for each RGBA pixel in place.
32pub fn premultiply_alpha_rgba8_in_place(pixels: &mut [u8]) {
33    for chunk in pixels.chunks_exact_mut(4) {
34        let a = chunk[3] as f32 / 255.0;
35        chunk[0] = (chunk[0] as f32 * a) as u8;
36        chunk[1] = (chunk[1] as f32 * a) as u8;
37        chunk[2] = (chunk[2] as f32 * a) as u8;
38    }
39}
40impl Texture {
41    /// Parse a texture color-space label and return the matching enum value.
42    pub fn parse_color_space(value: &str) -> Option<TextureColorSpace> {
43        match value.to_ascii_lowercase().as_str() {
44            "srgb" => Some(TextureColorSpace::Srgb),
45            "linear" => Some(TextureColorSpace::Linear),
46            _ => None,
47        }
48    }
49    /// Load a texture with sRGB color space by default.
50    pub fn load<P: AsRef<Path>>(
51        path: P,
52        textures: &mut SlotMap<TextureKey, TextureData>,
53    ) -> EngineResult<Self> {
54        Self::load_with_color_space(path, textures, TextureColorSpace::Srgb)
55    }
56    /// Load a texture from disk, premultiply alpha, and store it in the texture pool.
57    pub fn load_with_color_space<P: AsRef<Path>>(
58        path: P,
59        textures: &mut SlotMap<TextureKey, TextureData>,
60        color_space: TextureColorSpace,
61    ) -> EngineResult<Self> {
62        let img = ::image::open(&path).map_err(|e| {
63            let path_str = path.as_ref().display().to_string();
64            if path_str.contains("No such file") || matches!(e, ::image::ImageError::IoError(_)) {
65                EngineError::ResourceNotFound(format!("{}: {}", path_str, e))
66            } else {
67                EngineError::RenderError(format!("Failed to decode image '{}': {}", path_str, e))
68            }
69        })?;
70        let rgba = img.to_rgba8();
71        let (width, height) = rgba.dimensions();
72        let mut pixels = rgba.into_raw();
73        premultiply_alpha_rgba8_in_place(&mut pixels);
74        let key = textures.insert(TextureData {
75            pixels,
76            width,
77            height,
78            color_space,
79        });
80        log_msg!(debug, TX01_TEX_DECODED, "{}x{}", width, height);
81        Ok(Texture { key, width, height })
82    }
83    /// Create a texture from RGBA bytes using sRGB color space.
84    pub fn from_rgba(
85        width: u32,
86        height: u32,
87        pixels: Vec<u8>,
88        textures: &mut SlotMap<TextureKey, TextureData>,
89    ) -> EngineResult<Self> {
90        Self::from_rgba_with_color_space(width, height, pixels, textures, TextureColorSpace::Srgb)
91    }
92    /// Create a texture from RGBA bytes and store it in the texture pool.
93    pub fn from_rgba_with_color_space(
94        width: u32,
95        height: u32,
96        mut pixels: Vec<u8>,
97        textures: &mut SlotMap<TextureKey, TextureData>,
98        color_space: TextureColorSpace,
99    ) -> EngineResult<Self> {
100        let expected = (width * height * 4) as usize;
101        if pixels.len() != expected {
102            return Err(EngineError::RenderError(format!(
103                "Expected {} bytes for {}x{} RGBA, got {}",
104                expected,
105                width,
106                height,
107                pixels.len()
108            )));
109        }
110        premultiply_alpha_rgba8_in_place(&mut pixels);
111        let key = textures.insert(TextureData {
112            pixels,
113            width,
114            height,
115            color_space,
116        });
117        Ok(Texture { key, width, height })
118    }
119}

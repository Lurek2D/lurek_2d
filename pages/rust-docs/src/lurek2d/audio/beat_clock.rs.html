1//! Implements musical time tracking that maps wall-clock progression to beats, bars, and pulses.
2//! Supports tempo and meter changes while preserving coherent phase continuity over runtime updates.
3//! Provides tap-tempo and quantized scheduling utilities for rhythm-aware gameplay coordination.
4//! Applies latency and swing parameters to shape musical timing feel without audio-thread coupling.
5//! Exposes deterministic query surfaces for beat index, measure position, and subdivision boundaries.
6//! Keeps timing logic pure and playback-agnostic so multiple systems can consume one clock source.
7//! Serves rhythm, sequencing, and procedural trigger systems that require stable musical time.
8//! Functions as the temporal backbone for Lua callbacks aligned to musical structure.
9
10/// Runtime options used when creating a beat clock.
11#[derive(Clone, Copy, Debug)]
12pub struct BeatClockOpts {
13    /// Default subdivision used by division-optional query helpers.
14    pub subdivision: u32,
15    /// Swing amount in range [0, 0.5].

18    pub latency: f64,
19}

173    /// Advance the clock by `dt` seconds. Returns a list of whole-beat crossings
174    /// (as beat numbers) that occurred during this step.
175    pub fn tick(&mut self, dt: f64) -> Vec<f64> {
176        if !self.running || dt <= 0.0 {
177            return Vec::new();
178        }
179        let before = self.beat_total();
180        let ramp_active_before = self.target_bpm.is_some();
181        let bpm_before = self.bpm;
182        self.advance_ramp(dt);
183        let bpm_for_step = if ramp_active_before {
184            (bpm_before + self.bpm) * 0.5
185        } else {
186            self.bpm
187        };
188        self.elapsed_secs += dt;
189        self.beat_position += dt * bpm_for_step / 60.0;
190        let after = self.beat_total();
191
192        // Collect whole beats that were crossed
193        let start = before.floor() + 1.0;
194        let mut crossings = Vec::new();
195        let mut b = start;
196        while b <= after {
197            crossings.push(b);
198            b += 1.0;
199        }
200        crossings
201    }
202
203    /// Advance the clock and emit beat/bar transition events.
204    pub fn update(&mut self, dt: f64) -> BeatClockEvents {
205        let before_bar = self.get_bar().floor() as u32;
206        let crossings = self.tick(dt);
207        let mut events = BeatClockEvents::default();
208        if let Some(beat) = crossings.last() {
209            events.new_beat = Some(*beat as u32);
210        }
211        let after_bar = self.get_bar().floor() as u32;
212        if after_bar > before_bar {
213            events.new_bar = Some(after_bar);
214        }
215        events
216    }

218    /// Returns the current musical position.
219    pub fn position(&self) -> BeatPosition {
220        let beat = self.beat_total();
221        let bar = (beat / self.beats_per_bar as f64).floor() as u64;
222        let beat_in_bar = (beat.floor() as u64 % self.beats_per_bar as u64) as u32;
223        let phase = beat - beat.floor();
224        BeatPosition { beat, bar, beat_in_bar, phase }
225    }

242    /// Return the current phase in [0, 1) for a division of the beat.
243    pub fn get_phase(&self, division: u32) -> f64 {
244        let div = division.max(1);
245        let scaled = self.effective_beat() * div as f64;
246        let phase = scaled - scaled.floor();
247        if phase < 0.0 { phase + 1.0 } else { phase }
248    }

250    /// Return whether the clock is on a division grid within `tolerance` seconds.
251    pub fn is_on_beat(&self, division: u32, tolerance: f64) -> bool {
252        let (_, err_seconds) = self.nearest_beat(division);
253        err_seconds.abs() <= tolerance.max(0.0)
254    }
255
256    /// Return nearest grid beat and signed timing error in seconds.
257    pub fn nearest_beat(&self, division: u32) -> (f64, f64) {
258        let div = division.max(1) as f64;
259        let current = self.effective_beat();
260        let nearest = (current * div).round() / div;
261        let err_beats = current - nearest;
262        let err_seconds = err_beats * self.seconds_per_beat();
263        (nearest, err_seconds)
264    }
265
266    /// Return seconds until the next division boundary.
267    pub fn beat_time_remaining(&self, division: u32) -> f64 {
268        let div = division.max(1) as f64;
269        let current = self.effective_beat() * div;
270        let next = current.floor() + 1.0;
271        (next - current) * (self.seconds_per_beat() / div)
272    }

274    /// Judge a hit against the nearest beat at `division` with additional `hit_offset` seconds.
275    pub fn judge(&self, division: u32, hit_offset: f64) -> JudgementResult {
276        let (_, err_seconds) = self.nearest_beat(division);
277        let total = err_seconds + hit_offset;
278        let abs = total.abs();
279        if abs <= self.judgement_windows.perfect {
280            JudgementResult::Perfect(total)
281        } else if abs <= self.judgement_windows.great {
282            JudgementResult::Great(total)
283        } else if abs <= self.judgement_windows.good {
284            JudgementResult::Good(total)
285        } else {
286            JudgementResult::Miss(total)
287        }
288    }

374    /// Record a tap for tap-tempo estimation. After at least 2 taps the estimated BPM
375    /// is applied immediately. Returns the new BPM, or 0.0 when not enough taps yet.
376    pub fn tap(&mut self, wall_time_secs: f64) -> f64 {
377        self.tap_times.push(wall_time_secs);
378        // Keep only the last 8 taps
379        if self.tap_times.len() > 8 {
380            self.tap_times.remove(0);
381        }
382        if self.tap_times.len() < 2 {
383            return 0.0;
384        }
385        let n = self.tap_times.len();
386        let interval = (self.tap_times[n - 1] - self.tap_times[0]) / (n - 1) as f64;
387        let new_bpm = 60.0 / interval.max(f64::EPSILON);
388        self.bpm = new_bpm.clamp(20.0, 400.0);
389        self.start_bpm = self.bpm;
390        self.target_bpm = None;
391        self.ramp_t = 0.0;
392        self.ramp_dur = 0.0;
393        self.bpm
394    }

396    /// Schedule a one-shot beat callback at `beat`. Returns `true` when the beat
397    /// is in the future relative to the current position.
398    pub fn schedule_at(&mut self, beat: f64) -> bool {
399        if beat <= self.beat_total() {
400            return false;
401        }
402        if let Err(pos) = self.scheduled.binary_search_by(|b| b.partial_cmp(&beat).unwrap()) {
403            self.scheduled.insert(pos, beat);
404        }
405        true
406    }

408    /// Drain all scheduled beats that are now in the past. Call after `tick`.
409    pub fn drain_fired(&mut self) -> Vec<f64> {
410        let now = self.beat_total();
411        let (fired, remaining): (Vec<f64>, Vec<f64>) =
412            self.scheduled.drain(..).partition(|&b| b <= now);
413        self.scheduled = remaining;
414        fired
415    }

429    /// Quantise `beat` to the nearest `grid` beat grid (e.g. 0.25 for 16th notes).
430    pub fn quantise(beat: f64, grid: f64) -> f64 {
431        if grid <= 0.0 { return beat; }
432        (beat / grid).round() * grid
433    }

435    /// Return 1-based step indices crossed in `(before, after]` for a beat division.
436    pub fn crossed_steps(before: f64, after: f64, division: u32) -> Vec<i64> {
437        if after <= before {
438            return Vec::new();
439        }
440        let div = division.max(1) as f64;
441        let start = (before * div).floor() as i64 + 1;
442        let end = (after * div).floor() as i64;
443        if end < start {
444            return Vec::new();
445        }
446        (start..=end).collect()
447    }

449    fn advance_ramp(&mut self, dt: f64) {
450        let Some(target) = self.target_bpm else {
451            return;
452        };
453        if self.ramp_dur <= 0.0 {
454            self.bpm = target;
455            self.start_bpm = target;
456            self.target_bpm = None;
457            self.ramp_t = 0.0;
458            self.ramp_dur = 0.0;
459            return;
460        }
461        self.ramp_t = (self.ramp_t + dt).min(self.ramp_dur);
462        let t = (self.ramp_t / self.ramp_dur).clamp(0.0, 1.0);
463        self.bpm = self.start_bpm + (target - self.start_bpm) * t;
464        if self.ramp_t >= self.ramp_dur {
465            self.bpm = target;
466            self.start_bpm = target;
467            self.target_bpm = None;
468            self.ramp_t = 0.0;
469            self.ramp_dur = 0.0;
470        }
471    }

473    fn effective_beat(&self) -> f64 {
474        self.beat_total() + self.latency * self.bpm / 60.0
475    }

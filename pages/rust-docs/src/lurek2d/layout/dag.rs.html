1//! Provides staged layered layout for directed graphs where flow direction and rank readability are primary goals.
2//! Organizes nodes into bands, reorders local neighborhoods to reduce crossings, and then assigns stable screen coordinates.
3//! Applies spacing and margin policy from shared layout config so outputs align with other module strategies.
4//! Prefers deterministic structure over visual drift to keep dependency and progression maps legible across updates.
5//! Serves graph-like UI flows that need clear upstream-downstream interpretation without manual node placement.
6
7use super::types::*;
8use std::collections::{HashMap, HashSet, VecDeque};
9
10/// Lays out a directed acyclic graph using a layered approach.
11///
12/// Nodes are assigned to layers based on longest path from sources,
13/// then positioned within each layer to minimize crossings.
14pub fn layout_dag(
15    nodes: &[LayoutNode],
16    edges: &[LayoutEdge],
17    config: &LayoutConfig,
18) -> LayoutResult {
19    if nodes.is_empty() {
20        return LayoutResult::new(Vec::new());
21    }
22
23    let layers = assign_layers(nodes, edges);
24    let ordered_layers = reduce_crossings(&layers, edges);
25    let positioned = assign_coordinates(nodes, &ordered_layers, config);
26
27    LayoutResult::new(positioned)
28}

30/// Assigns layers based on longest path from source nodes.
31fn assign_layers(nodes: &[LayoutNode], edges: &[LayoutEdge]) -> Vec<Vec<NodeId>> {
32    let node_ids: HashSet<NodeId> = nodes.iter().map(|n| n.id).collect();
33    let mut in_degree: HashMap<NodeId, usize> = node_ids.iter().map(|&id| (id, 0)).collect();
34    let mut adj: HashMap<NodeId, Vec<NodeId>> = node_ids.iter().map(|&id| (id, Vec::new())).collect();
35
36    for edge in edges {
37        if node_ids.contains(&edge.from) && node_ids.contains(&edge.to) {
38            adj.entry(edge.from).or_default().push(edge.to);
39            *in_degree.entry(edge.to).or_insert(0) += 1;
40        }
41    }
42
43    let mut layer_of: HashMap<NodeId, usize> = HashMap::new();
44    let mut queue: VecDeque<NodeId> = in_degree
45        .iter()
46        .filter(|(_, &deg)| deg == 0)
47        .map(|(&id, _)| id)
48        .collect();
49
50    for &id in &queue {
51        layer_of.insert(id, 0);
52    }
53
54    while let Some(node) = queue.pop_front() {
55        let current_layer = layer_of[&node];
56        let neighbors = adj.get(&node).cloned().unwrap_or_default();
57        for next in neighbors {
58            let new_layer = current_layer + 1;
59            let entry = layer_of.entry(next).or_insert(0);
60            if new_layer > *entry {
61                *entry = new_layer;
62            }
63            let deg = in_degree.get_mut(&next).unwrap();
64            *deg -= 1;
65            if *deg == 0 {
66                queue.push_back(next);
67            }
68        }
69    }
70
71    let max_layer = layer_of.values().copied().max().unwrap_or(0);
72    let mut layers: Vec<Vec<NodeId>> = vec![Vec::new(); max_layer + 1];
73    for (&id, &layer) in &layer_of {
74        layers[layer].push(id);
75    }
76    layers
77}

79/// Reduces crossings using barycenter heuristic (single pass).
80fn reduce_crossings(layers: &[Vec<NodeId>], edges: &[LayoutEdge]) -> Vec<Vec<NodeId>> {
81    let mut result = layers.to_vec();
82
83    for i in 1..result.len() {
84        let prev_positions: HashMap<NodeId, usize> = result[i - 1]
85            .iter()
86            .enumerate()
87            .map(|(pos, &id)| (id, pos))
88            .collect();
89
90        let mut barycenters: Vec<(NodeId, f64)> = result[i]
91            .iter()
92            .map(|&node| {
93                let parents: Vec<usize> = edges
94                    .iter()
95                    .filter(|e| e.to == node)
96                    .filter_map(|e| prev_positions.get(&e.from).copied())
97                    .collect();
98                let bc = if parents.is_empty() {
99                    0.0
100                } else {
101                    parents.iter().sum::<usize>() as f64 / parents.len() as f64
102                };
103                (node, bc)
104            })
105            .collect();
106
107        barycenters.sort_by(|a, b| a.1.partial_cmp(&b.1).unwrap_or(std::cmp::Ordering::Equal));
108        result[i] = barycenters.into_iter().map(|(id, _)| id).collect();
109    }
110
111    result
112}

114/// Assigns x/y coordinates to nodes based on their layer and position.
115fn assign_coordinates(
116    nodes: &[LayoutNode],
117    layers: &[Vec<NodeId>],
118    config: &LayoutConfig,
119) -> Vec<LayoutNode> {
120    let node_map: HashMap<NodeId, &LayoutNode> = nodes.iter().map(|n| (n.id, n)).collect();
121    let mut result = Vec::new();
122
123    for (layer_idx, layer) in layers.iter().enumerate() {
124        let y = config.margin + layer_idx as f64 * config.v_spacing;
125        let mut x = config.margin;
126
127        for &node_id in layer {
128            if let Some(&original) = node_map.get(&node_id) {
129                let mut node = original.clone();
130                node.x = x;
131                node.y = y;
132                x += node.width + config.h_spacing;
133                result.push(node);
134            }
135        }
136    }
137
138    result
139}

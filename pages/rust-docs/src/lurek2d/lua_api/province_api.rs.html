23/// Resolves a province asset path against the active game directory when the path is relative.
24fn resolve_game_path(state: &Rc<RefCell<SharedState>>, path: &str) -> String {
25    let st = state.borrow();
26    let p = Path::new(path);
27    if p.is_absolute() {
28        p.to_string_lossy().into_owned()
29    } else {
30        st.game_dir.join(p).to_string_lossy().into_owned()
31    }
32}
33/// Parses optional marker sanitization settings from a Lua options table.

35    let mut out = MarkerSanitizeOptions::default();
36    if let Some(t) = opts {
37        if let Some(v) = t.get::<_, Option<u8>>("capital_min").ok().flatten() {

48        }
49        if let Some(v) = t.get::<_, Option<u32>>("search_radius").ok().flatten() {
50            out.search_radius = v;

56fn parse_border_pair_flags_from_lua(value: LuaValue) -> LuaResult<BorderPairFlags> {
57    match value {
58        LuaValue::Nil => Ok(BorderPairFlags::empty()),
59        LuaValue::String(s) => {
60            let token = s.to_str()?.to_lowercase();
61            let bit = BorderPairFlags::parse_token(token.as_str()).ok_or_else(|| {
62                LuaError::RuntimeError(format!("invalid border flag '{}'", token))
63            })?;
64            Ok(BorderPairFlags::from_bits(bit))
65        }
66        LuaValue::Table(t) => {
67            let mut flags = BorderPairFlags::empty();
68            for value in t.sequence_values::<String>() {
69                let token = value?.to_lowercase();
70                let bit = BorderPairFlags::parse_token(token.as_str()).ok_or_else(|| {
71                    LuaError::RuntimeError(format!("invalid border flag '{}'", token))
72                })?;
73                flags.insert_bits(bit);
74            }
75            Ok(flags)
76        }
77        _ => Err(LuaError::RuntimeError(
78            "flags must be a string or array of strings".to_string(),
79        )),
80    }
81}

83fn parse_border_pair_style_from_lua(style: &LuaTable) -> LuaResult<BorderPairStyle> {
84    let thickness = style
85        .get::<_, Option<f32>>("thickness")?
86        .unwrap_or(1.0)
87        .max(1.0);
88    let color = if let Some(color_tbl) = style.get::<_, Option<LuaTable>>("color")? {
89        Some([
90            color_tbl.get::<_, Option<f32>>(1)?.unwrap_or(1.0),
91            color_tbl.get::<_, Option<f32>>(2)?.unwrap_or(1.0),
92            color_tbl.get::<_, Option<f32>>(3)?.unwrap_or(1.0),
93            color_tbl.get::<_, Option<f32>>(4)?.unwrap_or(1.0),
94        ])
95    } else {
96        None
97    };
98    let flags_value = style.get::<_, Option<LuaValue>>("flags")?.unwrap_or(LuaValue::Nil);
99    let flags = parse_border_pair_flags_from_lua(flags_value)?;
100    Ok(BorderPairStyle {
101        color,
102        thickness,
103        flags,
104    })
105}

121impl LuaProvinceRegistry {
122    /// Runs a closure with the current province registry or returns a Lua runtime error when missing.
123    fn with_registry<R>(&self, f: impl FnOnce(&ProvinceRegistry) -> R) -> LuaResult<R> {
124        let st = self.state.borrow();
125        let reg = st.province_registries.get(&self.name).ok_or_else(|| {
126            LuaError::RuntimeError(format!("province registry '{}' not found", self.name))
127        })?;
128        Ok(f(reg))
129    }
130    fn with_registry_mut<R>(&self, f: impl FnOnce(&mut ProvinceRegistry) -> R) -> LuaResult<R> {
131        let mut st = self.state.borrow_mut();
132        let reg = st.province_registries.get_mut(&self.name).ok_or_else(|| {
133            LuaError::RuntimeError(format!("province registry '{}' not found", self.name))
134        })?;
135        Ok(f(reg))
136    }

418        // -- findRoute --
419        /// Finds a route between two provinces using BFS or Dijkstra when `cost_fn` is supplied.
420        /// @param | from_id | integer | Start province id.
421        /// @param | to_id | integer | Target province id.
422        /// @param | cost_fn | function? | Optional cost callback `fn(from_id, to_id) -> number`.
423        /// @return | table | Array of province ids from start to target; nil when unreachable.
424        methods.add_method("findRoute", |lua, this, (from_id, to_id, cost_fn): (u32, u32, Option<LuaFunction>)| {
425            let (pairs, all_ids) = this.with_registry(|r| {
426                let pairs: Vec<(u32, u32)> = r
427                    .adjacency_pairs()
428                    .into_iter()
429                    .map(|(a, b)| (a.0, b.0))
430                    .collect();
431                let ids: Vec<u32> = r.province_ids().into_iter().map(|id| id.0).collect();
432                (pairs, ids)
433            })?;
434
435            let adjacency = routing::build_adjacency_map(&pairs);
436            let has_from = all_ids.contains(&from_id);
437            let has_to = all_ids.contains(&to_id);
438            if !has_from || !has_to {
439                return Ok(LuaValue::Nil);
440            }
441
442            let route = if let Some(f) = cost_fn {
443                let mut edge_cost: HashMap<(u32, u32), f64> = HashMap::new();
444                for (a, b) in pairs {
445                    let ab = f.call::<_, Option<f64>>((a, b))?.unwrap_or(1.0).max(0.000_001);
446                    let ba = f.call::<_, Option<f64>>((b, a))?.unwrap_or(1.0).max(0.000_001);
447                    edge_cost.insert((a, b), ab);
448                    edge_cost.insert((b, a), ba);
449                }
450                routing::find_route_dijkstra(&adjacency, from_id, to_id, &|a, b| {
451                    edge_cost.get(&(a, b)).copied().unwrap_or(1.0)
452                })
453            } else {
454                routing::find_route_bfs(&adjacency, from_id, to_id)
455            };
456
457            if let Some(path) = route {
458                let out = lua.create_table()?;
459                for (i, id) in path.into_iter().enumerate() {
460                    out.set(i + 1, id)?;
461                }
462                Ok(LuaValue::Table(out))
463            } else {
464                Ok(LuaValue::Nil)
465            }
466        });
467        // -- findRoutes --
468        /// Finds routes for a batch of `{from, to}` pairs.
469        /// @param | pairs | table | Array of `{from=integer, to=integer}` tables.
470        /// @param | cost_fn | function? | Optional cost callback `fn(from_id, to_id) -> number?`.
471        /// @return | table | Array of route arrays (or nil for unreachable entries).
472        methods.add_method("findRoutes", |lua, this, (pairs_tbl, cost_fn): (LuaTable, Option<LuaFunction>)| {
473            let (pairs, all_ids) = this.with_registry(|r| {
474                let pairs: Vec<(u32, u32)> = r
475                    .adjacency_pairs()
476                    .into_iter()
477                    .map(|(a, b)| (a.0, b.0))
478                    .collect();
479                let ids: Vec<u32> = r.province_ids().into_iter().map(|id| id.0).collect();
480                (pairs, ids)
481            })?;
482            let adjacency = routing::build_adjacency_map(&pairs);
483
484            let mut edge_cost: HashMap<(u32, u32), f64> = HashMap::new();
485            if let Some(ref f) = cost_fn {
486                for &(a, b) in &pairs {
487                    let ab = f.call::<_, Option<f64>>((a, b))?.unwrap_or(1.0).max(0.000_001);
488                    let ba = f.call::<_, Option<f64>>((b, a))?.unwrap_or(1.0).max(0.000_001);
489                    edge_cost.insert((a, b), ab);
490                    edge_cost.insert((b, a), ba);
491                }
492            }
493
494            let out = lua.create_table()?;
495            for (i, pair_val) in pairs_tbl.sequence_values::<LuaTable>().enumerate() {
496                let t = pair_val?;
497                let from_id: u32 = t.get("from")?;
498                let to_id: u32 = t.get("to")?;
499
500                let route = if !all_ids.contains(&from_id) || !all_ids.contains(&to_id) {
501                    None
502                } else if cost_fn.is_some() {
503                    routing::find_route_dijkstra(&adjacency, from_id, to_id, &|a, b| {
504                        edge_cost.get(&(a, b)).copied().unwrap_or(1.0)
505                    })
506                } else {
507                    routing::find_route_bfs(&adjacency, from_id, to_id)
508                };
509
510                if let Some(path) = route {
511                    let row = lua.create_table()?;
512                    for (j, id) in path.into_iter().enumerate() {
513                        row.set(j + 1, id)?;
514                    }

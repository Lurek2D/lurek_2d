20/// Lua-side handle for a named globe stored inside a shared registry.
21pub struct LuaGlobe {
22    /// Shared globe registry containing all named globes.
23    reg: Arc<Mutex<GlobeRegistry>>,
24    /// Registry key for the globe represented by this Lua handle.
25    name: String,
26    #[allow(dead_code)]
27    /// Shared runtime state retained for future renderer integration.
28    state: Rc<RefCell<SharedState>>,
29}
30impl LuaGlobe {
31    /// Reads the named globe from the registry and maps it to a Lua result.
32    fn with<R>(&self, f: impl FnOnce(&Globe) -> R) -> LuaResult<R> {
33        let guard = self
34            .reg
35            .lock()
36            .map_err(|e| mlua::Error::RuntimeError(format!("globe registry lock poisoned: {e}")))?;
37        guard
38            .get(&self.name)
39            .map(f)
40            .ok_or_else(|| mlua::Error::RuntimeError(format!("globe '{}' not found", self.name)))
41    }
42    /// Mutates the named globe in the registry and maps it to a Lua result.
43    fn with_mut<R>(&self, f: impl FnOnce(&mut Globe) -> R) -> LuaResult<R> {
44        let mut guard = self
45            .reg
46            .lock()
47            .map_err(|e| mlua::Error::RuntimeError(format!("globe registry lock poisoned: {e}")))?;
48        guard
49            .get_mut(&self.name)
50            .map(f)
51            .ok_or_else(|| mlua::Error::RuntimeError(format!("globe '{}' not found", self.name)))
52    }
53}

62            let id: u32 = p.get("id")?;
63            let ct: LuaTable = p.get("centroid")?;
64            let centroid = (ct.get::<_, f32>(1)?, ct.get::<_, f32>(2)?);
65            let verts_tbl: LuaTable = p.get::<_, LuaTable>("vertices").map_err(|_| {
66                mlua::Error::RuntimeError(format!("province {id}: 'vertices' field is required"))
67            })?;
68            let mut vertices = Vec::new();
69            for vt in verts_tbl.sequence_values::<LuaTable>() {
70                let vt = vt?;
71                vertices.push((vt.get::<_, f32>(1)?, vt.get::<_, f32>(2)?));
72            }
73            let neighbors: Vec<RegionId> = p
74                .get::<_, LuaTable>("neighbors")
75                .map(|t| t.sequence_values::<u32>().map(|r| r.map(RegionId)).collect::<LuaResult<Vec<_>>>())

360                this.with(|g| {
361                    let cx = g.camera.screen_cx;
362                    let cy = g.camera.screen_cy;
363                    let dx = sx - cx;
364                    let dy = sy - cy;
365                    for i in 1..=n {
366                        let t = i as f32 / n as f32;
367                        let px = cx + dx * t;
368                        let py = cy + dy * t;
369                        if let Some(hit) = g.pick_screen(px, py) {
370                            return Some(hit.region_id.0);
371                        }
372                    }

798                })
799            },
800        );
801        // -- removeArc --
802        /// Removes an arc by id. This method is available to Lua scripts.
803        /// @param | id | integer | Arc id.
804        /// @return | boolean | True when an arc was removed.
805        methods.add_method_mut("removeArc", |_, this, id: u32| {

807        });
808        // -- getName --
809        /// Returns the registry name of this globe.
810        /// @return | string | Globe registry name.
811        methods.add_method("getName", |_, this, ()| Ok(this.name.clone()));
812        methods.add_meta_method(LuaMetaMethod::ToString, |_, this, ()| {

836/// Provides Lua methods for managing globe registry entries.
837impl LuaUserData for LuaGlobeRegistry {
838    fn add_methods<'lua, M: LuaUserDataMethods<'lua, Self>>(methods: &mut M) {
839        // -- new --
840        /// Creates a named globe with optional specification fields.
841        /// @param | name | string | Globe registry name.
842        /// @param | spec_tbl | table? | Globe specification table.
843        /// @return | LGlobe | New globe handle.
844        methods.add_method_mut(
845            "new",
846            |_, this, (name, spec_tbl): (String, Option<LuaTable>)| {
847                let spec = parse_globe_spec(spec_tbl);
848                {
849                    let mut guard = this.reg.lock().map_err(|e| {
850                        mlua::Error::RuntimeError(format!("registry lock poisoned: {e}"))
851                    })?;
852                    guard.create(name.clone(), spec);
853                }
854                Ok(LuaGlobe {
855                    reg: this.reg.clone(),
856                    name,
857                    state: this.state.clone(),
858                })
859            },
860        );
861        // -- get --
862        /// Returns a globe handle by registry name.
863        /// @param | name | string | Globe registry name.
864        /// @return | LGlobe | Globe handle, or nil when no globe exists with that name.
865        methods.add_method("get", |_, this, name: String| {
866            let exists = {
867                let guard = this.reg.lock().map_err(|e| {
868                    mlua::Error::RuntimeError(format!("registry lock poisoned: {e}"))
869                })?;
870                guard.get(&name).is_some()
871            };
872            if exists {
873                Ok(Some(LuaGlobe {
874                    reg: this.reg.clone(),
875                    name,
876                    state: this.state.clone(),
877                }))
878            } else {
879                Ok(None)
880            }
881        });
882        // -- remove --
883        /// Removes a globe from the registry by name.
884        /// @param | name | string | Globe registry name.
885        /// @return | boolean | True when a globe was removed.
886        methods.add_method_mut("remove", |_, this, name: String| {
887            let mut guard = this
888                .reg
889                .lock()
890                .map_err(|e| mlua::Error::RuntimeError(format!("registry lock poisoned: {e}")))?;
891            Ok(guard.remove(&name).is_some())
892        });
893        // -- names --
894        /// Returns all globe names currently stored in this registry.
895        /// @return | string[] | Globe names.
896        methods.add_method("names", |lua, this, ()| {
897            let guard = this
898                .reg
899                .lock()
900                .map_err(|e| mlua::Error::RuntimeError(format!("registry lock poisoned: {e}")))?;
901            let names = guard.names();
902            let t = lua.create_table()?;
903            for (i, n) in names.iter().enumerate() {
904                t.set(i + 1, n.clone())?;
905            }
906            Ok(t)
907        });
908        // -- type --
909        /// Returns the Lua-visible type name for this globe registry handle.
910        /// @return | string | The string `LGlobeRegistry`.
911        methods.add_method("type", |_, _, ()| Ok("LGlobeRegistry"));
912        // -- typeOf --
913        /// Returns whether this registry handle matches a supported type name.
914        /// @param | name | string | Type name to compare against `LGlobeRegistry` and `Object`.
915        /// @return | boolean | True when the supplied type name matches this handle.
916        methods.add_method("typeOf", |_, _, name: String| {
917            Ok(name == "LGlobeRegistry" || name == "LObject")
918        });
919    }
920}
921/// Parses optional Lua table fields into a globe specification.
922fn parse_globe_spec(tbl: Option<LuaTable>) -> GlobeSpec {
923    let mut spec = GlobeSpec::default();
924    if let Some(t) = tbl {
925        if let Ok(v) = t.get::<_, f32>("radius") {
926            spec.radius = v;
927        }
928        if let Ok(v) = t.get::<_, f32>("axial_tilt_deg") {
929            spec.axial_tilt_deg = v;
930        }
931        if let Ok(v) = t.get::<_, f32>("rotation_deg") {
932            spec.rotation_deg = v;
933        }
934        if let Ok(v) = t.get::<_, f32>("time_of_day") {
935            spec.time_of_day = v;
936        }
937        if let Ok(v) = t.get::<_, bool>("render_borders") {
938            spec.render_borders = v;
939        }
940        if let Ok(v) = t.get::<_, f32>("border_width") {
941            spec.border_width = v;
942        }
943        if let Ok(v) = t.get::<_, f32>("ambient") {
944            spec.ambient = v;
945        }
946        if let Ok(v) = t.get::<_, f32>("auto_rotation_deg_per_sec") {
947            spec.auto_rotation_deg_per_sec = v;
948        }
949        if let Ok(v) = t.get::<_, f32>("atmosphere_width") {
950            spec.atmosphere_width = v.max(0.0);
951        }
952        if let Ok(v) = t.get::<_, u8>("border_smoothing_passes") {
953            spec.border_smoothing_passes = v;
954        }
955    }
956    spec
957}

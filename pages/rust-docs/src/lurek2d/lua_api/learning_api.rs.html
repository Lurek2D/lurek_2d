13/// Lua handle for a Q-learning table with configurable exploration and learning parameters.
14#[derive(Clone)]
15pub(crate) struct LuaQLearner {
16    /// Shared Q-learning model containing Q-values, episode counters, and tuning parameters.
17    pub(crate) inner: Rc<RefCell<QLearner>>,
18}
19impl LuaUserData for LuaQLearner {
20    fn add_methods<'lua, M: LuaUserDataMethods<'lua, Self>>(methods: &mut M) {
21        // -- chooseAction --
22        /// Chooses an action for a one-based state index using the learner's exploration policy.
23        /// @param | state | integer | One-based state index.
24        /// @return | integer | One-based chosen action index.
25        methods.add_method("chooseAction", |_, this, state: usize| {
26            Ok(this.inner.borrow().choose_action(state.saturating_sub(1)) + 1)
27        });
28        // -- bestAction --
29        /// Returns the highest-valued action for a one-based state index without exploration.
30        /// @param | state | integer | One-based state index.
31        /// @return | integer | One-based best action index.
32        methods.add_method("bestAction", |_, this, state: usize| {
33            Ok(this.inner.borrow().best_action(state.saturating_sub(1)) + 1)
34        });
35        // -- learn --
36        /// Applies one Q-learning update from a transition and reward.
37        /// @param | state | integer | One-based previous state index.
38        /// @param | action | integer | One-based action index taken in the previous state.
39        /// @param | reward | number | Reward received for the transition.
40        /// @param | next_state | integer | One-based next state index.
41        methods.add_method(
42            "learn",
43            |_, this, (state, action, reward, next_state): (usize, usize, f64, usize)| {
44                this.inner.borrow_mut().learn(
45                    state.saturating_sub(1),
46                    action.saturating_sub(1),
47                    reward,
48                    next_state.saturating_sub(1),
49                );
50                Ok(())
51            },
52        );
53        // -- getQValue --
54        /// Returns the stored Q-value for a one-based state and action pair.
55        /// @param | state | integer | One-based state index.
56        /// @param | action | integer | One-based action index.
57        /// @return | number | Current Q-value.
58        methods.add_method("getQValue", |_, this, (state, action): (usize, usize)| {
59            Ok(this
60                .inner
61                .borrow()
62                .get_q(state.saturating_sub(1), action.saturating_sub(1)))
63        });
64        // -- setQValue --
65        /// Sets the stored Q-value for a one-based state and action pair.
66        /// @param | state | integer | One-based state index.
67        /// @param | action | integer | One-based action index.
68        /// @param | value | number | Q-value to store.
69        methods.add_method(
70            "setQValue",
71            |_, this, (state, action, value): (usize, usize, f64)| {
72                this.inner.borrow_mut().set_q(
73                    state.saturating_sub(1),
74                    action.saturating_sub(1),
75                    value,
76                );
77                Ok(())
78            },
79        );
80        // -- endEpisode --
81        /// Decays epsilon and increments the episode count.
82        methods.add_method("endEpisode", |_, this, ()| {
83            this.inner.borrow_mut().end_episode();
84            Ok(())
85        });
86        // -- getEpisodeCount --
87        /// Returns the total number of episodes completed so far.
88        /// @return | integer | Episode count.
89        methods.add_method("getEpisodeCount", |_, this, ()| {
90            Ok(this.inner.borrow().episode_count)
91        });
92        // -- getStateCount --
93        /// Returns the number of states represented by this learner.
94        /// @return | integer | State count.
95        methods.add_method("getStateCount", |_, this, ()| {
96            Ok(this.inner.borrow().state_count)
97        });
98        // -- getActionCount --
99        /// Returns the number of actions represented by this learner.
100        /// @return | integer | Action count.
101        methods.add_method("getActionCount", |_, this, ()| {
102            Ok(this.inner.borrow().action_count)
103        });
104        // -- setLearningRate --
105        /// Sets the Q-learning alpha learning rate.
106        /// @param | v | number | Learning rate used by future updates.
107        methods.add_method("setLearningRate", |_, this, v: f64| {
108            this.inner.borrow_mut().alpha = v;
109            Ok(())
110        });
111        // -- getLearningRate --
112        /// Returns the Q-learning alpha learning rate.
113        /// @return | number | Current learning rate.
114        methods.add_method("getLearningRate", |_, this, ()| {
115            Ok(this.inner.borrow().alpha)
116        });
117        // -- setDiscountFactor --
118        /// Sets the Q-learning gamma discount factor.
119        /// @param | v | number | Discount factor used by future updates.
120        methods.add_method("setDiscountFactor", |_, this, v: f64| {
121            this.inner.borrow_mut().gamma = v;
122            Ok(())
123        });
124        // -- getDiscountFactor --
125        /// Returns the Q-learning gamma discount factor.
126        /// @return | number | Current discount factor.
127        methods.add_method("getDiscountFactor", |_, this, ()| {
128            Ok(this.inner.borrow().gamma)
129        });
130        // -- setExplorationRate --
131        /// Sets the exploration rate used by action selection.
132        /// @param | v | number | Exploration probability for future `chooseAction` calls.
133        methods.add_method("setExplorationRate", |_, this, v: f64| {
134            this.inner.borrow_mut().epsilon = v;
135            Ok(())
136        });
137        // -- getExplorationRate --
138        /// Returns the exploration rate used by action selection.
139        /// @return | number | Current exploration rate.
140        methods.add_method("getExplorationRate", |_, this, ()| {
141            Ok(this.inner.borrow().epsilon)
142        });
143        // -- setExplorationDecay --
144        /// Sets the exploration decay multiplier applied across episodes.
145        /// @param | v | number | Exploration decay multiplier.
146        methods.add_method("setExplorationDecay", |_, this, v: f64| {
147            this.inner.borrow_mut().epsilon_decay = v;
148            Ok(())
149        });
150        // -- getExplorationDecay --
151        /// Returns the exploration decay multiplier.
152        /// @return | number | Current exploration decay multiplier.
153        methods.add_method("getExplorationDecay", |_, this, ()| {
154            Ok(this.inner.borrow().epsilon_decay)
155        });
156        // -- serialize --
157        /// Serializes the Q-learner state to a JSON string.
158        /// @return | string | JSON representation of this learner.
159        methods.add_method("serialize", |_, this, ()| {
160            Ok(this.inner.borrow().serialize())
161        });
162        // -- deserialize --
163        /// Replaces the Q-learner state from a JSON string.
164        /// @param | json | string | JSON data previously produced by `serialize`.
165        methods.add_method("deserialize", |_, this, json: String| {
166            this.inner
167                .borrow_mut()
168                .deserialize(&json)
169                .map_err(LuaError::RuntimeError)?;
170            Ok(())
171        });
172        // -- type --
173        /// Returns the Lua-visible type name for this Q-learner handle.
174        /// @return | string | The string `LQLearner`.
175        methods.add_method("type", |_, _, ()| Ok("LQLearner"));
176        // -- typeOf --
177        /// Returns whether this Q-learner handle matches a supported type name.
178        /// @param | name | string | Type name to compare against `LQLearner` and `Object`.
179        /// @return | boolean | True when the supplied type name matches this handle.
180        methods.add_method("typeOf", |_, _, name: String| {
181            Ok(name == "LQLearner" || name == "LObject")
182        });
183        // -- predict --
184        /// Alias for `chooseAction`. Selects an action for the given one-based state using the learner's policy.
185        /// @param | state | integer | One-based state index.
186        /// @return | integer | One-based chosen action index.
187        methods.add_method("predict", |_, this, state: usize| {
188            Ok(this.inner.borrow().choose_action(state.saturating_sub(1)) + 1)
189        });
190    }

199impl LuaUserData for LuaNeuralNet {
200    fn add_methods<'lua, M: LuaUserDataMethods<'lua, Self>>(methods: &mut M) {
201        // -- addLayer --
202        /// Adds a neural network layer with an activation function.
203        /// @param | inputs | integer | Input count for the layer.
204        /// @param | outputs | integer | Output count for the layer.
205        /// @param | activation | string | Activation name such as `relu`, `sigmoid`, `tanh`, `linear`, or `softmax`.
206        methods.add_method_mut(
207            "addLayer",
208            |_, this, (inputs, outputs, activation): (usize, usize, String)| {
209                let act = Activation::from_str(&activation);
210                this.inner.borrow_mut().add_layer(inputs, outputs, act);
211                Ok(())
212            },
213        );
214        // -- forward --
215        /// Runs a forward pass and returns output values.
216        /// @param | input | table | Array of numeric input values.
217        /// @return | number[] | Numeric output values.
218        methods.add_method("forward", |lua, this, input: Vec<f32>| {
219            let out = this.inner.borrow().forward(&input);
220            let t = lua.create_table()?;
221            for (i, v) in out.into_iter().enumerate() {
222                t.raw_set(i + 1, v)?;
223            }
224            Ok(t)
225        });
226        // -- setWeights --
227        /// Replaces the network weights from a flat numeric array.
228        /// @param | weights | table | Flat array of numeric weights in engine layer order.
229        /// @return | boolean | True when the supplied weight slice matches the network shape.
230        methods.add_method_mut("setWeights", |_, this, weights: Vec<f32>| {
231            Ok(this.inner.borrow_mut().set_weights(&weights))
232        });
233        // -- getWeights --
234        /// Returns the network weights as a flat numeric array.
235        /// @return | number[] | Numeric weights in engine layer order.
236        methods.add_method("getWeights", |lua, this, ()| {
237            let w = this.inner.borrow().get_weights();
238            let t = lua.create_table()?;
239            for (i, v) in w.into_iter().enumerate() {
240                t.raw_set(i + 1, v)?;
241            }
242            Ok(t)
243        });
244        // -- paramCount --
245        /// Returns the total number of trainable parameters.
246        /// @return | integer | Parameter count.
247        methods.add_method("paramCount", |_, this, ()| {
248            Ok(this.inner.borrow().param_count() as i64)
249        });
250        // -- layerCount --
251        /// Returns the number of layers in the network.
252        /// @return | integer | Layer count.
253        methods.add_method("layerCount", |_, this, ()| {
254            Ok(this.inner.borrow().layer_count() as i64)
255        });
256        // -- type --
257        /// Returns the Lua-visible type name for this neural network handle.
258        /// @return | string | The string `LNeuralNet`.
259        methods.add_method("type", |_, _, ()| Ok("LNeuralNet"));
260        // -- typeOf --

263        /// @return | boolean | True when the supplied type name matches this handle.
264        methods.add_method("typeOf", |_, _, name: String| {
265            Ok(name == "LNeuralNet" || name == "LObject")
266        });
267        // -- predict --
268        /// Alias for `forward`. Runs a forward pass and returns output values.

329            }

498/// A uniform model wrapper that delegates `predict` to any supported learning model type.
499///
500/// Wraps one of `LuaQLearner`, `LuaNeuralNet`, or `LuaBandit` so callers can use a
501/// single `predict(...)` call without knowing the concrete model type at call sites.
502#[derive(Clone)]
503pub(crate) enum LuaModel {
504    QLearner(LuaQLearner),
505    NeuralNet(LuaNeuralNet),
506    Bandit(LuaBandit),
507}
508impl LuaUserData for LuaModel {
509    fn add_methods<'lua, M: LuaUserDataMethods<'lua, Self>>(methods: &mut M) {
510        // -- predict --
511        /// Runs the wrapped model's prediction. Delegates to `chooseAction`, `forward`, or `select`
512        /// depending on the wrapped type. Input is interpreted as `integer` for QLearner/Bandit
513        /// and as a `table` of numbers for NeuralNet.
514        /// @param | input | any | State index (integer) for QLearner/Bandit, or number array table for NeuralNet.
515        /// @return | integer, table | Action index for QLearner/Bandit, or number-array table for NeuralNet.
516        methods.add_method_mut("predict", |lua, this, input: LuaValue| {
517            match this {
518                LuaModel::QLearner(q) => {
519                    let state: usize = lua.unpack(input)?;
520                    Ok(lua.pack(q.inner.borrow().choose_action(state.saturating_sub(1)) + 1)?)
521                }
522                LuaModel::NeuralNet(n) => {
523                    let v: Vec<f32> = lua.unpack(input)?;
524                    let out = n.inner.borrow().forward(&v);
525                    let t = lua.create_table()?;
526                    for (i, val) in out.into_iter().enumerate() {
527                        t.raw_set(i + 1, val)?;
528                    }
529                    Ok(lua.pack(t)?)
530                }
531                LuaModel::Bandit(b) => {
532                    Ok(lua.pack(b.inner.borrow_mut().select() as i64)?)
533                }
534            }
535        });
536        // -- type --
537        /// Returns this wrapper's stable type name `"LModel"`.
538        /// @return | string | The string `LModel`.
539        methods.add_method("type", |_, _, ()| Ok("LModel"));
540        // -- typeOf --
541        /// Returns whether this model wrapper matches a supported type name.
542        /// @param | name | string | Type name to compare against `LModel` and `Object`.
543        /// @return | boolean | True when the supplied type name matches this wrapper.
544        methods.add_method("typeOf", |_, _, name: String| {
545            Ok(name == "LModel" || name == "LObject")
546        });
547    }
548}

559    action_space: SpaceSpec,
560    normalize_mean: Option<Vec<f32>>,
561    normalize_std: Option<Vec<f32>>,
562    time_limit: Option<u32>,
563    step_count: u32,
564    inner_env: Option<Rc<RefCell<LuaEnv>>>,
565}
566
567impl LuaEnv {
568    fn call_reset(&mut self, lua: &Lua) -> LuaResult<Vec<f32>> {
569        if let Some(ref inner) = self.inner_env.clone() {
570            return inner.borrow_mut().call_reset(lua);
571        }
572        let key = self
573            .reset_fn
574            .as_ref()
575            .ok_or_else(|| LuaError::RuntimeError("LEnv: no reset function defined".into()))?;
576        let f: LuaFunction = lua.registry_value(key)?;
577        let result: LuaTable = f.call(())?;
578        let mut obs = Vec::with_capacity(result.raw_len());
579        for i in 1..=result.raw_len() {
580            let v: f64 = result.raw_get(i).unwrap_or(0.0);
581            obs.push(v as f32);
582        }
583        Ok(obs)
584    }
585
586    fn call_step<'lua>(
587        &mut self,
588        lua: &'lua Lua,
589        action: LuaValue<'lua>,
590    ) -> LuaResult<(Vec<f32>, f32, bool, LuaTable<'lua>)> {
591        if let Some(ref inner) = self.inner_env.clone() {
592            return inner.borrow_mut().call_step(lua, action);
593        }
594        let key = self
595            .step_fn
596            .as_ref()
597            .ok_or_else(|| LuaError::RuntimeError("LEnv: no step function defined".into()))?;
598        let f: LuaFunction = lua.registry_value(key)?;
599        let result: LuaTable = f.call(action)?;
600        let obs_tbl: LuaTable = result.raw_get(1)?;
601        let reward: f64 = result.raw_get(2).unwrap_or(0.0);
602        let done: bool = result.raw_get(3).unwrap_or(false);
603        let info: LuaTable = result
604            .raw_get(4)
605            .unwrap_or_else(|_| lua.create_table().expect("table"));
606        let mut obs = Vec::with_capacity(obs_tbl.raw_len());
607        for i in 1..=obs_tbl.raw_len() {
608            let v: f64 = obs_tbl.raw_get(i).unwrap_or(0.0);
609            obs.push(v as f32);
610        }
611        Ok((obs, reward as f32, done, info))
612    }
613
614    fn apply_normalize(&self, obs: &mut [f32]) {
615        if let (Some(mean), Some(std)) = (&self.normalize_mean, &self.normalize_std) {
616            for (i, v) in obs.iter_mut().enumerate() {
617                let m = mean.get(i).copied().unwrap_or(0.0);
618                let s = std.get(i).copied().unwrap_or(1.0);
619                let s = if s == 0.0 { 1.0 } else { s };
620                *v = (*v - m) / s;
621            }
622        }
623    }
624
625    fn obs_to_table<'lua>(lua: &'lua Lua, obs: Vec<f32>) -> LuaResult<LuaTable<'lua>> {
626        let t = lua.create_table()?;
627        for (i, v) in obs.into_iter().enumerate() {
628            t.raw_set(i + 1, v as f64)?;
629        }
630        Ok(t)
631    }
632
633    fn space_to_table<'lua>(lua: &'lua Lua, space: &SpaceSpec) -> LuaResult<LuaTable<'lua>> {
634        let t = lua.create_table()?;
635        let shape_tbl = lua.create_table()?;
636        for (i, &v) in space.shape.iter().enumerate() {
637            shape_tbl.raw_set(i + 1, v)?;
638        }
639        t.raw_set("shape", shape_tbl)?;
640        let low_tbl = lua.create_table()?;
641        for (i, &v) in space.low.iter().enumerate() {
642            low_tbl.raw_set(i + 1, v as f64)?;
643        }
644        t.raw_set("low", low_tbl)?;
645        let high_tbl = lua.create_table()?;
646        for (i, &v) in space.high.iter().enumerate() {
647            high_tbl.raw_set(i + 1, v as f64)?;
648        }
649        t.raw_set("high", high_tbl)?;
650        if space.n > 0 {
651            t.raw_set("n", space.n)?;
652        }
653        Ok(t)
654    }
655}

657impl LuaUserData for LuaEnv {
658    fn add_methods<'lua, M: LuaUserDataMethods<'lua, Self>>(methods: &mut M) {
659        // -- reset --
660        /// Resets the environment and returns the initial observation.
661        /// @return | number[] | Initial observation vector.
662        methods.add_method_mut("reset", |lua, this, ()| {
663            this.step_count = 0;
664            let mut obs = this.call_reset(lua)?;
665            this.apply_normalize(&mut obs);
666            Self::obs_to_table(lua, obs)
667        });
668        // -- step --
669        /// Advances the environment one step.
670        /// @param | action | any | Action to apply (integer or table depending on action space).
671        /// @return | number[] | Next observation vector.
672        /// @return | number | Reward for this step.
673        /// @return | boolean | Whether the episode has ended.
674        /// @return | table | Extra info table.
675        methods.add_method_mut(
676            "step",
677            |lua, this, action: LuaValue| {
678                let (mut obs, reward, mut done, info) = this.call_step(lua, action)?;
679                this.apply_normalize(&mut obs);
680                this.step_count += 1;
681                if let Some(limit) = this.time_limit {
682                    if this.step_count >= limit {
683                        done = true;
684                    }
685                }
686                let obs_tbl = Self::obs_to_table(lua, obs)?;
687                Ok((obs_tbl, reward as f64, done, info))
688            },
689        );
690        // -- obsSpace --
691        /// Returns the observation space descriptor.
692        /// @return | table | Observation space with shape, low, high fields.
693        methods.add_method("obsSpace", |lua, this, ()| {
694            Self::space_to_table(lua, &this.obs_space)
695        });
696        // -- actionSpace --
697        /// Returns the action space descriptor.
698        /// @return | table | Action space with shape/low/high or n fields.
699        methods.add_method("actionSpace", |lua, this, ()| {
700            Self::space_to_table(lua, &this.action_space)
701        });
702        // -- type --
703        /// Returns this environment wrapper's type name `"LEnv"`.
704        /// @return | string | The string `LEnv`.
705        methods.add_method("type", |_, _, ()| Ok("LEnv"));
706        // -- typeOf --
707        /// Returns whether this env handle matches a supported type name.
708        /// @param | name | string | Type name to compare against `LEnv` and `Object`.
709        /// @return | boolean | True when the supplied type name matches this handle.
710        methods.add_method("typeOf", |_, _, name: String| {
711            Ok(name == "LEnv" || name == "LObject")
712        });
713    }

773/// Parses a `SpaceSpec` from a Lua table.
774fn parse_space_spec(tbl: &LuaTable) -> LuaResult<SpaceSpec> {
775    let shape_tbl: Option<LuaTable> = tbl.raw_get("shape").ok();
776    let mut shape = Vec::new();
777    if let Some(st) = shape_tbl {
778        for i in 1..=st.raw_len() {
779            let v: u32 = st.raw_get(i).unwrap_or(1);
780            shape.push(v);
781        }
782    }
783    let low_tbl: Option<LuaTable> = tbl.raw_get("low").ok();
784    let mut low = Vec::new();
785    if let Some(lt) = low_tbl {
786        for i in 1..=lt.raw_len() {
787            let v: f64 = lt.raw_get(i).unwrap_or(0.0);
788            low.push(v as f32);
789        }
790    }
791    let high_tbl: Option<LuaTable> = tbl.raw_get("high").ok();
792    let mut high = Vec::new();
793    if let Some(ht) = high_tbl {
794        for i in 1..=ht.raw_len() {
795            let v: f64 = ht.raw_get(i).unwrap_or(1.0);
796            high.push(v as f32);
797        }
798    }
799    let n: u32 = tbl.raw_get("n").unwrap_or(0);
800    Ok(SpaceSpec { shape, low, high, n })
801}

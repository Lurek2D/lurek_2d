12/// Lua-side handle for a single cached asset entry.
13///
14/// Wraps the numeric handle ID and a shared reference to the cache so that
15/// methods can query or mutate ref counts without an external state reference.
16#[derive(Clone)]
17pub struct LuaAssetHandle {
18    id: u64,
19    cache: Rc<RefCell<AssetCache>>,
20}

22impl LuaUserData for LuaAssetHandle {
23    fn add_methods<'lua, M: LuaUserDataMethods<'lua, Self>>(methods: &mut M) {
24        // -- type --
25        /// Returns the Lua-visible type name for this asset handle.
26        /// @return | string | The string `LAssetHandle`.
27        methods.add_method("type", |_, _, ()| Ok("LAssetHandle"));
28
29        // -- typeOf --
30        /// Returns whether this handle matches a supported type name.
31        /// @param | name | string | Type name to compare against `LAssetHandle` and `LObject`.
32        /// @return | boolean | True when the supplied type name matches this handle.
33        methods.add_method("typeOf", |_, _, name: String| {
34            Ok(name == "LAssetHandle" || name == "LObject")
35        });
36    }

39fn load_text_content(path: &str, asset_type: &AssetType, context: &str) -> LuaResult<Option<String>> {
40    if asset_type.is_text_like() {
41        let content = std::fs::read_to_string(path).map_err(|e| {
42            LuaError::RuntimeError(format!(
43                "{}: cannot read \"{}\" (type: {}): {}",
44                context,
45                path,
46                asset_type.as_str(),
47                e
48            ))
49        })?;
50        Ok(Some(content))
51    } else {
52        if !std::path::Path::new(path).exists() {
53            return Err(LuaError::RuntimeError(format!(
54                "{}: file not found: \"{}\"",
55                context,
56                path
57            )));
58        }
59        Ok(None)
60    }
61}

63fn apply_load_opts(cache: &Rc<RefCell<AssetCache>>, id: u64, opts: Option<LuaTable>) -> LuaResult<()> {
64    if let Some(opts) = opts {
65        if let Ok(name) = opts.get::<_, String>("name") {
66            cache.borrow_mut().set_name(id, name);
67        }
68        if let Ok(group) = opts.get::<_, String>("group") {
69            cache.borrow_mut().set_group(id, group);
70        }
71        if let Ok(tags_tbl) = opts.get::<_, LuaTable>("tags") {
72            let len = tags_tbl.raw_len();
73            for i in 1..=len {
74                if let Ok(tag) = tags_tbl.get::<_, String>(i) {
75                    cache.borrow_mut().add_tag(id, &tag);
76                }
77            }
78        }
79    }
80    Ok(())
81}

95fn get_handle_entry(cache: &Rc<RefCell<AssetCache>>, id: u64, context: &str) -> LuaResult<crate::asset::AssetEntry> {
96    let borrow = cache.borrow();

100    Ok(crate::asset::AssetEntry {
101        path: entry.path.clone(),
102        asset_type: entry.asset_type.clone(),
103        ref_count: entry.ref_count,
104        text_content: entry.text_content.clone(),
105        name: entry.name.clone(),
106        group: entry.group.clone(),
107        tags: entry.tags.clone(),
108    })

111fn resolve_asset_value(lua: &Lua, entry: crate::asset::AssetEntry) -> LuaResult<LuaValue<'_>> {
112    match entry.asset_type {
113        AssetType::Text
114        | AssetType::Toml
115        | AssetType::Json
116        | AssetType::Obj
117        | AssetType::Shader
118        | AssetType::Lua => match entry.text_content {
119            Some(s) => Ok(LuaValue::String(lua.create_string(&s)?)),
120            None => Ok(LuaValue::Nil),
121        },
122        AssetType::Image => {
123            let image_tbl: LuaTable = lua.globals().get::<_, LuaTable>("lurek")?.get("image")?;
124            let load_fn: LuaFunction = image_tbl.get("loadImage")?;
125            load_fn.call(entry.path)
126        }
127        AssetType::Font => {
128            let font_tbl: LuaTable = lua.globals().get::<_, LuaTable>("lurek")?.get("font")?;
129            let load_fn: LuaFunction = font_tbl.get("load")?;
130            load_fn.call((entry.path, 16i64))
131        }
132        AssetType::Audio | AssetType::Music => {
133            let audio_tbl: LuaTable = lua.globals().get::<_, LuaTable>("lurek")?.get("audio")?;
134            let new_source_fn: LuaFunction = audio_tbl.get("newSource")?;
135            new_source_fn.call(entry.path)
136        }
137        AssetType::Unknown(_) => Ok(LuaValue::Nil),
138    }
139}

141fn preload_assets(cache: &Rc<RefCell<AssetCache>>, paths: LuaTable, callback: LuaFunction) -> LuaResult<()> {
142    let count = paths.raw_len() as i64;
143    for i in 1..=count {
144        let row: LuaTable = paths.get(i)?;
145
146        let path: String = row
147            .get::<_, Option<String>>(1)?
148            .or_else(|| row.get::<_, Option<String>>("path").ok().flatten())
149            .ok_or_else(|| {
150                LuaError::RuntimeError(
151                    "lurek.asset.preload: each entry must have a path".into(),
152                )
153            })?;
154
155        let type_str: String = row
156            .get::<_, Option<String>>(2)?
157            .or_else(|| row.get::<_, Option<String>>("type").ok().flatten())
158            .ok_or_else(|| {
159                LuaError::RuntimeError(
160                    "lurek.asset.preload: each entry must have a type".into(),
161                )
162            })?;
163
164        let asset_type = AssetType::from_type_str(&type_str);
165        let text_content = load_text_content(&path, &asset_type, "lurek.asset.preload")?;
166        cache.borrow_mut().register(path, asset_type, text_content);
167        callback.call::<_, ()>((i, count))?;
168    }
169
170    callback.call::<_, ()>((LuaValue::Nil, LuaValue::Nil))?;
171    Ok(())
172}

176/// Registers `lurek.asset.*` functions into the `lurek` table.
177pub fn register(
178    lua: &Lua,
179    lurek: &LuaTable,
180    _state: Rc<RefCell<SharedState>>,
181) -> LuaResult<()> {
182    let cache: Rc<RefCell<AssetCache>> = Rc::new(RefCell::new(AssetCache::new()));
183
184    let asset_tbl = lua.create_table()?;
185
186    // â”€â”€â”€ load â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€

188    // -- load --
189    /// Loads and caches an asset by path and type, returning a ref-counted handle.
190    /// Supported types: `image`, `font`, `audio`, `music`, `text`, `toml`, `json`,
191    /// `obj`, `shader`, `lua`. Text-like types read and cache the file content;
192    /// binary types (`image`, `font`, `audio`, `music`) store the path reference only
193    /// and are resolved lazily when `lurek.asset.get` is called.
194    /// @param | path | string | Filesystem path to the asset file.
195    /// @param | asset_type | string | Asset type string; see above for valid values.
196    /// @param | opts | table? | Optional metadata: `{name, group, tags}`.
197    /// @return | LAssetHandle | Handle that keeps the asset alive in the cache.
198    let load_cache = cache.clone();
199    asset_tbl.set(
200        "load",
201        lua.create_function(
202            move |_lua, (path, type_str, opts): (String, String, Option<LuaTable>)| {
203                let asset_type = AssetType::from_type_str(&type_str);
204                let text_content = load_text_content(&path, &asset_type, "lurek.asset.load")?;
205                let id = load_cache.borrow_mut().register(path, asset_type, text_content);
206                apply_load_opts(&load_cache, id, opts)?;
207                Ok(LuaAssetHandle {
208                    id,
209                    cache: load_cache.clone(),
210                })
211            },
212        )?,
213    )?;
214
215    // â”€â”€â”€ unload â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
216
217    // -- unload --
218    /// Decrements the ref count for a cached asset; removes the entry when it reaches zero.
219    /// @param | handle | LAssetHandle | Asset handle to release.
220    /// @return | nil | No value is returned.
221    asset_tbl.set(
222        "unload",
223        lua.create_function(|_lua, handle: LuaAnyUserData| {
224            let h = handle.borrow::<LuaAssetHandle>()?;
225            h.cache.borrow_mut().dec_ref(h.id);
226            Ok(())
227        })?,
228    )?;
229
230    // â”€â”€â”€ get â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
231
232    // -- get --
233    /// Returns the underlying asset value for a cached handle.
234    /// Text-like types (`text`, `toml`, `json`, `obj`, `shader`, `lua`) return the
235    /// cached file content as a string. `image` calls `lurek.image.loadImage`,
236    /// `font` calls `lurek.font.load`, `audio` and `music` call `lurek.audio.newSource`.
237    /// @param | handle | LAssetHandle | Asset handle to retrieve.
238    /// @return | string | Source text for text-like asset types.
239    /// @overload | handle | LAssetHandle | table | Runtime object returned by image/font/audio loaders for binary types.
240    let get_cache = cache.clone();
241    asset_tbl.set(
242        "get",
243        lua.create_function(move |lua, handle: LuaAnyUserData| {
244            let h = handle.borrow::<LuaAssetHandle>()?;
245            let entry = get_handle_entry(&get_cache, h.id, "lurek.asset.get")?;
246            resolve_asset_value(lua, entry)
247        })?,
248    )?;
249
250    // â”€â”€â”€ preload â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
251
252    // -- preload --
253    /// Synchronously loads a batch of assets and fires `callback(loaded, total)` after each item.
254    /// Fires `callback(nil, nil)` when all items have been processed.
255    /// @param | paths | table | Array of `{path, type}` pairs (or `{path=â€¦, type=â€¦}` tables).
256    /// @param | callback | any | Function invoked as `callback(loaded, total)` per item; `callback(nil, nil)` on finish.
257    /// @return | nil | No value is returned.
258    let preload_cache = cache.clone();
259    asset_tbl.set(
260        "preload",
261        lua.create_function(move |_lua, (paths, callback): (LuaTable, LuaFunction)| {
262            preload_assets(&preload_cache, paths, callback)
263        })?,
264    )?;
265
266    // â”€â”€â”€ refcount â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
267
268    // -- refcount --
269    /// Returns the current ref count for a handle, or 0 when it is no longer loaded.
270    /// @param | handle | LAssetHandle | Asset handle to inspect.
271    /// @return | integer | Current reference count.
272    asset_tbl.set(
273        "refcount",
274        lua.create_function(|_lua, handle: LuaAnyUserData| {
275            let h = handle.borrow::<LuaAssetHandle>()?;
276            let count = h.cache.borrow().ref_count(h.id);
277            Ok(count)
278        })?,
279    )?;
280
281    // â”€â”€â”€ isLoaded â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
282
283    // -- isLoaded --
284    /// Returns true when the asset for the given handle is still in the cache.
285    /// @param | handle | LAssetHandle | Asset handle to check.
286    /// @return | boolean | True when the asset is still cached.
287    asset_tbl.set(
288        "isLoaded",
289        lua.create_function(|_lua, handle: LuaAnyUserData| {
290            let h = handle.borrow::<LuaAssetHandle>()?;
291            let loaded = h.cache.borrow().is_loaded(h.id);
292            Ok(loaded)
293        })?,
294    )?;
295
296    // â”€â”€â”€ stats â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
297
298    // -- stats --
299    /// Returns a snapshot table describing the current cache state.
300    /// @return | table | Table with `loaded`, `total_refs`, `types`, and `groups` fields.
301    /// @field | loaded | integer | Number of distinct assets currently cached.
302    /// @field | total_refs | integer | Sum of all ref counts across all cached assets.
303    /// @field | types | table | Per-type entry counts keyed by type string.
304    /// @field | groups | table | Sorted array of unique group labels in the cache.
305    let stats_cache = cache.clone();
306    asset_tbl.set(
307        "stats",
308        lua.create_function(move |lua, ()| {
309            let (loaded, total_refs, type_counts, groups) = {
310                let borrow = stats_cache.borrow();
311                let mut counts = HashMap::<String, usize>::new();
312                for (_, entry) in borrow.iter() {
313                    *counts
314                        .entry(entry.asset_type.as_str().to_string())
315                        .or_insert(0) += 1;
316                }
317                (borrow.loaded_count(), borrow.total_refs(), counts, borrow.unique_groups())
318            };
319
320            let out = lua.create_table()?;
321            out.set("loaded", loaded)?;
322            out.set("total_refs", total_refs)?;
323            let per_type = lua.create_table()?;
324            for (k, v) in &type_counts {
325                per_type.set(k.as_str(), *v)?;
326            }
327            out.set("types", per_type)?;
328            let groups_array = lua.create_table()?;
329            for (i, g) in groups.iter().enumerate() {
330                groups_array.set(i + 1, g.as_str())?;
331            }
332            out.set("groups", groups_array)?;
333            Ok(out)
334        })?,
335    )?;
336
337    // â”€â”€â”€ clear â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
338
339    // -- clear --
340    /// Removes all entries from the cache immediately, regardless of ref counts.
341    /// @return | nil | No value is returned.
342    let clear_cache = cache.clone();
343    asset_tbl.set(
344        "clear",
345        lua.create_function(move |_lua, ()| {
346            clear_cache.borrow_mut().clear();
347            Ok(())
348        })?,
349    )?;
350
351    // â”€â”€â”€ getPath â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
352
353    // -- getPath --
354    /// Returns the filesystem path for the asset associated with a handle.
355    /// @param | handle | LAssetHandle | Asset handle to inspect.
356    /// @return | string | Path that was passed to `lurek.asset.load`.
357    asset_tbl.set(
358        "getPath",
359        lua.create_function(|_lua, handle: LuaAnyUserData| {
360            let h = handle.borrow::<LuaAssetHandle>()?;
361            let path = h
362                .cache
363                .borrow()
364                .get(h.id)
365                .ok_or_else(|| {
366                    LuaError::RuntimeError("lurek.asset.getPath: handle not loaded".into())
367                })?
368                .path
369                .clone();
370            Ok(path)
371        })?,
372    )?;
373
374    // â”€â”€â”€ getType â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€

376    // -- getType --
377    /// Returns the type string for the asset associated with a handle.
378    /// @param | handle | LAssetHandle | Asset handle to inspect.
379    /// @return | string | Type string, for example `"image"`, `"audio"`, `"toml"`.
380    asset_tbl.set(
381        "getType",
382        lua.create_function(|_lua, handle: LuaAnyUserData| {
383            let h = handle.borrow::<LuaAssetHandle>()?;
384            let type_str = h
385                .cache
386                .borrow()
387                .get(h.id)
388                .ok_or_else(|| {
389                    LuaError::RuntimeError("lurek.asset.getType: handle not loaded".into())
390                })?
391                .asset_type
392                .as_str()
393                .to_string();
394            Ok(type_str)
395        })?,
396    )?;
397
398    // â”€â”€â”€ getInfo â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
399
400    // -- getInfo --
401    /// Returns a table containing all metadata for an asset handle.
402    /// @param | handle | LAssetHandle | Asset handle to inspect.
403    /// @return | table | Metadata table; see fields below.
404    /// @field | path | string | Filesystem path to the asset.
405    /// @field | type | string | Asset type string.
406    /// @field | name | string | Display name, or the path file-stem when none is set.
407    /// @field | group | string | Group label, or empty string when none is set.
408    /// @field | tags | table | Array of tag strings.
409    /// @field | refcount | integer | Current reference count.
410    let info_cache = cache.clone();
411    asset_tbl.set(
412        "getInfo",
413        lua.create_function(move |lua, handle: LuaAnyUserData| {

20fn one_based_to_zero_based(value: u32, label: &str) -> LuaResult<u32> {
21    value.checked_sub(1).ok_or_else(|| {
22        LuaError::RuntimeError(format!(
23            "{} must be >= 1 for one-based coordinates",
24            label
25        ))
26    })
27}

29/// Converts zero-based Rust waypoints into one-based Lua point tables.
30fn waypoints_to_lua<'a>(lua: &'a Lua, path: &[Waypoint]) -> LuaResult<LuaTable<'a>> {
31    let tbl = lua.create_table()?;
32    for (i, wp) in path.iter().enumerate() {
33        let entry = lua.create_table()?;
34        /// The 'x' field value exposed to Lua scripts.
35        entry.set("x", wp.x + 1)?;
36        /// The 'y' field value exposed to Lua scripts.
37        entry.set("y", wp.y + 1)?;
38        tbl.set(i + 1, entry)?;
39    }
40    Ok(tbl)
41}
42/// Converts one-based Lua point tables into zero-based Rust waypoints.
43fn lua_to_waypoints(tbl: &LuaTable) -> LuaResult<Vec<Waypoint>> {
44    let mut waypoints = Vec::new();
45    for pair in tbl.clone().sequence_values::<LuaTable>() {
46        let entry = pair?;
47        let x: u32 = entry.get("x")?;
48        let y: u32 = entry.get("y")?;
49        waypoints.push(Waypoint {
50            x: one_based_to_zero_based(x, "x")?,
51            y: one_based_to_zero_based(y, "y")?,
52        });
53    }
54    Ok(waypoints)
55}

1528/// Provides `newGoalMap` methods: addSource, bake, distanceAt, gradientAt, flee, floodFill, save, restore.
1529impl LuaUserData for LuaGoalMap {
1530    fn add_methods<'lua, M: LuaUserDataMethods<'lua, Self>>(methods: &mut M) {
1531        // -- addSource --
1532        /// Registers a source cell for this goal map. Coordinates are one-based.
1533        /// @param | x      | integer  | One-based column.
1534        /// @param | y      | integer  | One-based row.
1535        /// @param | weight | integer? | Relative weight (default 1). Lower = stronger pull.

1544        /// Replaces all registered source cells. Each entry must have x, y (one-based) and optional weight.
1545        /// @param | sources | table | Array of `{x, y, weight?}` tables.
1546        methods.add_method("setSources", |_, this, sources: LuaTable| {
1547            let mut gs = Vec::new();
1548            for v in sources.sequence_values::<LuaTable>() {
1549                let t = v?;
1550                let x: u32 = t.get("x")?;
1551                let y: u32 = t.get("y")?;
1552                let w: u32 = t.get::<_, Option<u32>>("weight")?.unwrap_or(1);
1553                gs.push(GoalSource {
1554                    x: one_based_to_zero_based(x, "x")?,
1555                    y: one_based_to_zero_based(y, "y")?,
1556                    weight: w.max(1),
1557                });
1558            }
1559            this.inner.borrow_mut().set_sources(gs);
1560            Ok(())
1561        });
1562
1563        // -- clearSources --
1564        /// Removes all registered source cells.
1565        methods.add_method("clearSources", |_, this, ()| {
1566            this.inner.borrow_mut().clear_sources();
1567            Ok(())
1568        });
1569
1570        // -- setBlocker --
1571        /// Sets a Lua predicate called during `bake` to determine blocked cells.
1572        /// @param | fn | function | `fn(x: integer, y: integer) -> boolean` (one-based).
1573        methods.add_method("setBlocker", |lua, this, f: LuaFunction| {
1574            if let Some(old) = this.blocker_key.borrow_mut().take() {
1575                lua.remove_registry_value(old)?;
1576            }
1577            *this.blocker_key.borrow_mut() = Some(lua.create_registry_value(f)?);
1578            Ok(())
1579        });
1580
1581        // -- bake --
1582        /// Runs multi-source Dijkstra to build the distance field using the registered blocker.
1583        methods.add_method("bake", |lua, this, ()| {
1584            let blocker_key = this.blocker_key.borrow();
1585            if let Some(ref key) = *blocker_key {
1586                let f: LuaFunction = lua.registry_value(key)?;
1587                let w = this.inner.borrow().width();
1588                let h = this.inner.borrow().height();
1589                // Pre-compute blocker table to avoid Lua calls inside tight Rust loop.
1590                let size = (w * h) as usize;
1591                let mut blocked = vec![false; size];
1592                for y in 0..h {
1593                    for x in 0..w {
1594                        let result: bool = f.call((x + 1, y + 1))?;
1595                        blocked[(y * w + x) as usize] = result;
1596                    }
1597                }
1598                this.inner.borrow_mut().bake(&|x, y| blocked[(y * w + x) as usize]);
1599            } else {
1600                this.inner.borrow_mut().bake(&|_, _| false);
1601            }
1602            Ok(())
1603        });

1661                pt.set("x", x + 1)?;

1663                pt.set("y", y + 1)?;

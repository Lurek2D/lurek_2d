1//! Provides globe region-loading workflows from TOML, raster grids, and generated Voronoi seed sources.
2//! Parses lightweight structured input into normalized region records with geometry and adjacency data.
3//! Converts intermediate builder state into shared globe region types used across the subsystem.
4//! Extracts bounds and neighbor hints from image-driven province maps for quick content bootstrapping.
5//! Handles primitive parsing and validation to keep load-time failures explicit and actionable.
6//! Supports both in-memory string input and file-based ingestion paths for tooling flexibility.
7//! Delivers the map-ingestion layer that seeds topology and rendering state for globe runtime use.
8
9use crate::globe::types::{Region, RegionId};
10use crate::province::province_grid::ProvinceGrid;
11use crate::math::voronoi::voronoi_from_points;
12use std::collections::HashMap;
13/// Parsed TOML region record before conversion into the shared region type.
14#[derive(Debug, Clone)]
15struct TomlRegion {
16    /// Region identifier.
17    id: RegionId,
18    /// Region centroid as latitude and longitude.
19    centroid: [f32; 2],
20    /// Region polygon vertices as latitude and longitude pairs.
21    vertices: Vec<[f32; 2]>,
22    /// Neighboring region ids.
23    neighbors: Vec<RegionId>,
24    /// Optional base RGBA color.
25    base_color: Option<[f32; 4]>,
26    /// Optional region texture name.
27    texture: Option<String>,
28    /// Arbitrary string attributes.
29    attrs: HashMap<String, String>,
30}
31/// Load regions from a TOML string or return a parse error.
32pub fn load_from_toml_str(src: &str) -> Result<Vec<Region>, String> {
33    let doc = parse_toml_region_list(src)?;
34    Ok(doc.into_iter().map(toml_region_to_region).collect())
35}
36/// Load regions from a TOML file path or return a parse or I/O error.
37pub fn load_from_toml_file(path: &str) -> Result<Vec<Region>, String> {
38    let src =
39        std::fs::read_to_string(path).map_err(|e| format!("cannot read '{}': {}", path, e))?;
40    load_from_toml_str(&src)
41}
42/// Convert a parsed TOML region into the shared region type.
43fn toml_region_to_region(tp: TomlRegion) -> Region {
44    let verts: Vec<(f32, f32)> = tp.vertices.iter().map(|v| (v[0], v[1])).collect();
45    let mut p = Region::new(tp.id, verts);
46    p.centroid = (tp.centroid[0], tp.centroid[1]);
47    p.neighbors = tp.neighbors;
48    p.base_color = tp.base_color.unwrap_or([0.5, 0.5, 0.5, 1.0]);
49    p.texture = tp.texture;
50    p.attrs = tp.attrs;
51    p
52}
53/// Parse a TOML region list using the local lightweight parser.
54fn parse_toml_region_list(src: &str) -> Result<Vec<TomlRegion>, String> {
55    let mut regions: Vec<TomlRegion> = Vec::new();
56    let mut current: Option<TomlRegionBuilder> = None;
57    let mut in_attrs = false;
58    for (line_no, raw_line) in src.lines().enumerate() {
59        let line = raw_line.trim();
60        if line.starts_with('#') || line.is_empty() {
61            continue;
62        }
63        if line == "[[province]]" {
64            if let Some(b) = current.take() {
65                regions.push(b.build().map_err(|e| format!("line {}: {}", line_no, e))?);
66            }
67            current = Some(TomlRegionBuilder::default());
68            in_attrs = false;
69            continue;
70        }
71        if line == "[province.attrs]" {
72            in_attrs = true;
73            continue;
74        }
75        if line.starts_with('[') {
76            in_attrs = false;
77            continue;
78        }
79        if let Some(b) = current.as_mut() {
80            if in_attrs {
81                parse_kv_string(line, |k, v| {
82                    b.attrs.insert(k, v);
83                });
84            } else {
85                parse_field(line, b).map_err(|e| format!("line {}: {}", line_no, e))?;
86            }
87        }
88    }
89    if let Some(b) = current {
90        regions.push(b.build()?);
91    }
92    Ok(regions)
93}
94/// Incremental TOML region builder used by the lightweight parser.
95#[derive(Default)]
96struct TomlRegionBuilder {
97    /// Optional region identifier.
98    id: Option<RegionId>,
99    /// Optional centroid pair.
100    centroid: Option<[f32; 2]>,
101    /// Collected region vertices.
102    vertices: Vec<[f32; 2]>,
103    /// Collected neighbor ids.
104    neighbors: Vec<RegionId>,
105    /// Optional base RGBA color.
106    base_color: Option<[f32; 4]>,
107    /// Optional texture name.
108    texture: Option<String>,
109    /// Collected string attributes.
110    attrs: HashMap<String, String>,
111}
112impl TomlRegionBuilder {
113    /// Finalize the builder into a TOML region or return a missing-field error.
114    fn build(self) -> Result<TomlRegion, String> {
115        Ok(TomlRegion {
116            id: self.id.ok_or("province missing 'id'")?,
117            centroid: self.centroid.unwrap_or([0.0, 0.0]),
118            vertices: self.vertices,
119            neighbors: self.neighbors,
120            base_color: self.base_color,
121            texture: self.texture,
122            attrs: self.attrs,
123        })
124    }
125}
126/// Parse one non-attribute TOML field into the region builder.
127fn parse_field(line: &str, b: &mut TomlRegionBuilder) -> Result<(), String> {
128    let eq = line
129        .find('=')
130        .ok_or_else(|| format!("no '=' in '{}'", line))?;
131    let key = line[..eq].trim();
132    let val = line[eq + 1..].trim();
133    match key {
134        "id" => {
135            b.id = Some(RegionId(parse_u32(val)?));
136        }
137        "centroid" => {
138            b.centroid = Some(parse_f32_pair(val)?);
139        }
140        "vertices" => {
141            b.vertices = parse_f32_pair_array(val)?;
142        }
143        "neighbors" => {
144            b.neighbors = parse_u32_array(val)?.into_iter().map(RegionId).collect();
145        }
146        "base_color" => {
147            b.base_color = Some(parse_f32_4(val)?);
148        }
149        "texture" => {
150            b.texture = Some(strip_quotes(val).to_string());
151        }
152        _ => {}
153    }
154    Ok(())
155}
156/// Parse one string key-value pair line and call the supplied sink.
157fn parse_kv_string(line: &str, mut f: impl FnMut(String, String)) {
158    if let Some(eq) = line.find('=') {
159        let k = line[..eq].trim().to_string();
160        let v = strip_quotes(line[eq + 1..].trim()).to_string();
161        f(k, v);
162    }
163}
164/// Parse a u32 literal or return a parse error.
165fn parse_u32(s: &str) -> Result<u32, String> {
166    s.trim().parse().map_err(|_| format!("not a u32: '{}'", s))
167}
168/// Parse a two-element float array or return a parse error.
169fn parse_f32_pair(s: &str) -> Result<[f32; 2], String> {
170    let inner = s.trim().trim_start_matches('[').trim_end_matches(']');
171    let parts: Vec<&str> = inner.split(',').collect();
172    if parts.len() != 2 {
173        return Err(format!("expected [f, f], got '{}'", s));
174    }
175    Ok([
176        parts[0]
177            .trim()
178            .parse()
179            .map_err(|_| format!("bad f32 '{}'", parts[0]))?,
180        parts[1]
181            .trim()
182            .parse()
183            .map_err(|_| format!("bad f32 '{}'", parts[1]))?,
184    ])
185}
186/// Parse a four-element float array or return a parse error.
187fn parse_f32_4(s: &str) -> Result<[f32; 4], String> {
188    let inner = s.trim().trim_start_matches('[').trim_end_matches(']');
189    let parts: Vec<&str> = inner.split(',').collect();
190    if parts.len() != 4 {
191        return Err(format!("expected 4 floats, got '{}'", s));
192    }
193    let mut out = [0.0f32; 4];
194    for (i, p) in parts.iter().enumerate() {
195        out[i] = p.trim().parse().map_err(|_| format!("bad f32 '{}'", p))?;
196    }
197    Ok(out)
198}
199/// Parse an array of two-element float arrays or return a parse error.
200fn parse_f32_pair_array(s: &str) -> Result<Vec<[f32; 2]>, String> {
201    let s = s.trim().trim_start_matches('[').trim_end_matches(']');
202    let mut out = Vec::new();
203    let mut depth = 0i32;
204    let mut start = 0usize;
205    for (i, c) in s.char_indices() {
206        match c {
207            '[' => {
208                if depth == 0 {
209                    start = i;
210                }
211                depth += 1;
212            }
213            ']' => {
214                depth -= 1;
215                if depth == 0 {
216                    out.push(parse_f32_pair(&s[start..=i])?);
217                }
218            }
219            _ => {}
220        }
221    }
222    Ok(out)
223}
224/// Parse a u32 array or return a parse error.
225fn parse_u32_array(s: &str) -> Result<Vec<u32>, String> {
226    let inner = s.trim().trim_start_matches('[').trim_end_matches(']');
227    if inner.trim().is_empty() {
228        return Ok(Vec::new());
229    }
230    inner
231        .split(',')
232        .map(|p| p.trim().parse().map_err(|_| format!("bad u32 '{}'", p)))
233        .collect()
234}
235/// Strip matching single or double quotes from a string literal.
236fn strip_quotes(s: &str) -> &str {
237    let s = s.trim();
238    if (s.starts_with('"') && s.ends_with('"')) || (s.starts_with('\'') && s.ends_with('\'')) {
239        &s[1..s.len() - 1]
240    } else {
241        s
242    }
243}
244/// Load regions from a PNG province grid or return a decode or I/O error.
245pub fn load_from_png_file(_path: &str) -> Result<Vec<Region>, String> {
246    let grid = ProvinceGrid::from_file(_path)?;
247    let width = grid.width().max(1);
248    let height = grid.height().max(1);
249    let mut bounds: HashMap<RegionId, (u32, u32, u32, u32)> = HashMap::new();
250    for y in 0..height {
251        for x in 0..width {
252            let raw_id = grid.get_at(x, y);
253            if raw_id == 0 {
254                continue;
255            }
256            let id = RegionId(raw_id);
257            let e = bounds.entry(id).or_insert((x, y, x, y));
258            e.0 = e.0.min(x);
259            e.1 = e.1.min(y);
260            e.2 = e.2.max(x);
261            e.3 = e.3.max(y);
262        }
263    }
264    let mut neighbors: HashMap<RegionId, Vec<RegionId>> = HashMap::new();
265    for (a, b, _) in grid.adjacencies() {
266        let ra = RegionId(*a);
267        let rb = RegionId(*b);
268        neighbors.entry(ra).or_default().push(rb);
269        neighbors.entry(rb).or_default().push(ra);
270    }
271    let mut out = Vec::with_capacity(bounds.len());
272    for (id, (min_x, min_y, max_x, max_y)) in bounds {
273        let to_lon = |x: u32| (x as f32 / width as f32) * 360.0 - 180.0;
274        let to_lat = |y: u32| 90.0 - (y as f32 / height as f32) * 180.0;
275        let lat0 = to_lat(min_y);
276        let lat1 = to_lat(max_y);
277        let lon0 = to_lon(min_x);
278        let lon1 = to_lon(max_x);
279        let vertices = vec![(lat0, lon0), (lat0, lon1), (lat1, lon1), (lat1, lon0)];
280        let centroid = ((lat0 + lat1) * 0.5, (lon0 + lon1) * 0.5);
281        let p = Region::with_data(
282            id,
283            centroid,
284            vertices,
285            neighbors.remove(&id).unwrap_or_default(),
286            [0.5, 0.5, 0.5, 1.0],
287        );
288        out.push(p);
289    }
290    Ok(out)
291}
292/// Generate approximate regions from Voronoi input points.
293pub fn generate_voronoi_provinces(points: &[(f32, f32)]) -> Vec<Region> {
294    if points.is_empty() {
295        return Vec::new();
296    }
297    let pts_xy: Vec<(f32, f32)> = points.iter().map(|(lat, lon)| (*lon, *lat)).collect();
298    let cells = voronoi_from_points(&pts_xy);
299    let mut out = Vec::with_capacity(cells.len());
300    for (i, cell) in cells.iter().enumerate() {
301        let id = RegionId((i + 1) as u32);
302        let mut vertices = Vec::with_capacity(cell.vertices.len().max(3));
303        if cell.vertices.is_empty() {
304            let (x, y) = cell.site;
305            vertices.push((y - 0.5, x - 0.5));
306            vertices.push((y - 0.5, x + 0.5));
307            vertices.push((y + 0.5, x));
308        } else {
309            for (x, y) in &cell.vertices {
310                let lat = (*y).clamp(-90.0, 90.0);
311                let lon = (*x).clamp(-180.0, 180.0);
312                vertices.push((lat, lon));
313            }
314        }
315        let centroid = points.get(i).copied().unwrap_or((0.0, 0.0));
316        out.push(Region::with_data(
317            id,
318            centroid,
319            vertices,
320            Vec::new(),
321            [0.45, 0.45, 0.5, 1.0],
322        ));
323    }
324    for i in 0..out.len() {
325        let (ilat, ilon) = out[i].centroid;
326        let mut nearest: Vec<(f32, RegionId)> = out
327            .iter()
328            .filter(|p| p.id != out[i].id)
329            .map(|p| {
330                let dlat = ilat - p.centroid.0;
331                let dlon = ilon - p.centroid.1;
332                (dlat * dlat + dlon * dlon, p.id)
333            })
334            .collect();
335        nearest.sort_by(|a, b| a.0.partial_cmp(&b.0).unwrap_or(std::cmp::Ordering::Equal));
336        out[i].neighbors = nearest.into_iter().take(4).map(|(_, id)| id).collect();
337    }
338    out

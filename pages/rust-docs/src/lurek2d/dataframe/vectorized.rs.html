1//! Implements typed vectorized column storage for high-throughput dataframe-style numeric processing.
2//! Supports float, integer, boolean, and text columns with optional validity-mask semantics.
3//! Provides scalar element-wise transforms across arithmetic and unary operation families.
4//! Executes binary column operations with dtype-aware coercion and compatibility checks.
5//! Computes reductions including sum, mean, min, max, variance, and related aggregate metrics.
6//! Generates comparison masks for predicate-style filtering over typed column values.
7//! Supports bidirectional conversion between vectorized frames and generic dataframe representations.
8//! Applies parallelized multi-column operations and reductions via rayon-backed execution paths.
9//! Handles explicit column casting between numeric and textual type domains.
10//! Preserves boolean-mask filtering behavior consistently across all supported column types.
11//! Balances performance-oriented storage layout with conversion interoperability requirements.
12//! Serves as the vectorized acceleration layer above core dataframe contracts.

18/// Hold typed columnar storage with optional validity mask.
19pub enum ColumnStore {
20    /// Store 64-bit floating point values.
21    Float64 {
22        /// Store row values.
23        data: Vec<f64>,
24        /// Store optional per-row validity flags.
25        validity: Option<Vec<bool>>,
26    },
27    /// Store 64-bit integer values.
28    Int64 {
29        /// Store row values.
30        data: Vec<i64>,
31        /// Store optional per-row validity flags.
32        validity: Option<Vec<bool>>,
33    },
34    /// Store boolean values.
35    Bool {
36        /// Store row values.
37        data: Vec<bool>,
38        /// Store optional per-row validity flags.
39        validity: Option<Vec<bool>>,
40    },
41    /// Store UTF-8 text values.
42    Text {
43        /// Store row values.
44        data: Vec<String>,
45        /// Store optional per-row validity flags.
46        validity: Option<Vec<bool>>,
47    },
48}
49impl ColumnStore {
50    /// Return static type name for this column variant.
51    pub fn dtype_name(&self) -> &'static str {
52        match self {
53            ColumnStore::Float64 { .. } => "float64",
54            ColumnStore::Int64 { .. } => "int64",
55            ColumnStore::Bool { .. } => "bool",
56            ColumnStore::Text { .. } => "text",
57        }
58    }
59    /// Return number of rows in this column.
60    pub fn len(&self) -> usize {
61        match self {
62            ColumnStore::Float64 { data, .. } => data.len(),
63            ColumnStore::Int64 { data, .. } => data.len(),
64            ColumnStore::Bool { data, .. } => data.len(),
65            ColumnStore::Text { data, .. } => data.len(),
66        }
67    }
68    /// Return true when this column has no rows.
69    pub fn is_empty(&self) -> bool {
70        self.len() == 0
71    }
72    /// Return true when row at index is valid according to validity mask.
73    pub fn is_valid(&self, i: usize) -> bool {
74        let validity = match self {
75            ColumnStore::Float64 { validity, .. } => validity,
76            ColumnStore::Int64 { validity, .. } => validity,
77            ColumnStore::Bool { validity, .. } => validity,
78            ColumnStore::Text { validity, .. } => validity,
79        };
80        validity
81            .as_ref()
82            .is_none_or(|v| *v.get(i).unwrap_or(&false))
83    }
84    /// Return valid f64 values for Float64 columns, skipping nil rows.
85    pub fn valid_f64s(&self) -> Option<Vec<f64>> {
86        if let ColumnStore::Float64 { data, validity } = self {
87            let vals: Vec<f64> = if let Some(v) = validity {
88                data.iter()
89                    .zip(v.iter())
90                    .filter_map(|(x, ok)| if *ok { Some(*x) } else { None })
91                    .collect()
92            } else {
93                data.clone()
94            };
95            Some(vals)
96        } else {
97            None
98        }
99    }
100    /// Filter rows by boolean mask and return new column.
101    pub fn filter(&self, mask: &[bool]) -> ColumnStore {
102        match self {
103            ColumnStore::Float64 { data, validity } => {
104                let new_data: Vec<f64> = data
105                    .iter()
106                    .zip(mask.iter())
107                    .filter_map(|(v, ok)| if *ok { Some(*v) } else { None })
108                    .collect();
109                let new_validity = validity.as_ref().map(|val| {
110                    val.iter()
111                        .zip(mask.iter())
112                        .filter_map(|(v, ok)| if *ok { Some(*v) } else { None })
113                        .collect()
114                });
115                ColumnStore::Float64 {
116                    data: new_data,
117                    validity: new_validity,
118                }
119            }
120            ColumnStore::Int64 { data, validity } => {
121                let new_data: Vec<i64> = data
122                    .iter()
123                    .zip(mask.iter())
124                    .filter_map(|(v, ok)| if *ok { Some(*v) } else { None })
125                    .collect();
126                let new_validity = validity.as_ref().map(|val| {
127                    val.iter()
128                        .zip(mask.iter())
129                        .filter_map(|(v, ok)| if *ok { Some(*v) } else { None })
130                        .collect()
131                });
132                ColumnStore::Int64 {
133                    data: new_data,
134                    validity: new_validity,
135                }
136            }
137            ColumnStore::Bool { data, validity } => {
138                let new_data: Vec<bool> = data
139                    .iter()
140                    .zip(mask.iter())
141                    .filter_map(|(v, ok)| if *ok { Some(*v) } else { None })
142                    .collect();
143                let new_validity = validity.as_ref().map(|val| {
144                    val.iter()
145                        .zip(mask.iter())
146                        .filter_map(|(v, ok)| if *ok { Some(*v) } else { None })
147                        .collect()
148                });
149                ColumnStore::Bool {
150                    data: new_data,
151                    validity: new_validity,
152                }
153            }
154            ColumnStore::Text { data, validity } => {
155                let new_data: Vec<String> = data
156                    .iter()
157                    .zip(mask.iter())
158                    .filter_map(|(v, ok)| if *ok { Some(v.clone()) } else { None })
159                    .collect();
160                let new_validity = validity.as_ref().map(|val| {
161                    val.iter()
162                        .zip(mask.iter())
163                        .filter_map(|(v, ok)| if *ok { Some(*v) } else { None })
164                        .collect()
165                });
166                ColumnStore::Text {
167                    data: new_data,
168                    validity: new_validity,
169                }
170            }
171        }
172    }
173}

339    /// Convert DataFrame to VecFrame by inferring column types.
340    pub fn from_dataframe(df: &DataFrame) -> VecFrame {
341        let mut vf = VecFrame::new();
342        for (col_idx, name) in df.columns().iter().enumerate() {
343            let raw = df.raw_data();
344            let cells = &raw[col_idx];
345            let store = infer_column(cells);
346            vf.column_names.push(name.clone());
347            vf.columns.push(store);
348        }
349        vf

351    /// Convert VecFrame back to DataFrame.
352    pub fn to_dataframe(&self) -> DataFrame {
353        let nrows = self.nrows();
354        let mut col_data: Vec<Vec<CellValue>> = Vec::with_capacity(self.columns.len());
355        for (i, store) in self.columns.iter().enumerate() {
356            let _ = i;
357            let cells: Vec<CellValue> = (0..nrows)
358                .map(|row| {
359                    if !store.is_valid(row) {
360                        return CellValue::Nil;
361                    }
362                    match store {
363                        ColumnStore::Float64 { data, .. } => CellValue::Number(data[row]),
364                        ColumnStore::Int64 { data, .. } => CellValue::Number(data[row] as f64),
365                        ColumnStore::Bool { data, .. } => CellValue::Bool(data[row]),
366                        ColumnStore::Text { data, .. } => CellValue::Text(data[row].clone()),
367                    }
368                })
369                .collect();
370            col_data.push(cells);
371        }
372        DataFrame::from_raw(self.column_names.clone(), col_data)
373    }
374    /// Resolve column name to zero-based index.
375    fn resolve(&self, col: &str) -> Result<usize, String> {
376        self.column_names
377            .iter()
378            .position(|n| n == col)
379            .ok_or_else(|| format!("column not found: {col}"))
380    }
381    /// Apply scalar operation to Float64 column in place.
382    pub fn col_scalar_op(&mut self, col: &str, op: ScalarOp, val: f64) -> Result<(), String> {
383        let idx = self.resolve(col)?;
384        if let ColumnStore::Float64 { data, validity } = &mut self.columns[idx] {
385            let vref = validity.as_ref();
386            match op {
387                ScalarOp::Add => {
388                    for (i, x) in data.iter_mut().enumerate() {
389                        if vref.is_none_or(|v| v[i]) {
390                            *x += val;
391                        }
392                    }
393                }
394                ScalarOp::Sub => {
395                    for (i, x) in data.iter_mut().enumerate() {
396                        if vref.is_none_or(|v| v[i]) {
397                            *x -= val;
398                        }
399                    }
400                }
401                ScalarOp::Mul => {
402                    for (i, x) in data.iter_mut().enumerate() {
403                        if vref.is_none_or(|v| v[i]) {
404                            *x *= val;
405                        }
406                    }
407                }
408                ScalarOp::Div => {
409                    if val == 0.0 {
410                        return Err("division by zero in col_scalar_op".into());
411                    }
412                    for (i, x) in data.iter_mut().enumerate() {
413                        if vref.is_none_or(|v| v[i]) {
414                            *x /= val;
415                        }
416                    }
417                }
418                ScalarOp::Abs => {
419                    for (i, x) in data.iter_mut().enumerate() {
420                        if vref.is_none_or(|v| v[i]) {
421                            *x = x.abs();
422                        }
423                    }
424                }
425                ScalarOp::Sqrt => {
426                    for (i, x) in data.iter_mut().enumerate() {
427                        if vref.is_none_or(|v| v[i]) {
428                            *x = x.sqrt();
429                        }
430                    }
431                }
432                ScalarOp::Floor => {
433                    for (i, x) in data.iter_mut().enumerate() {
434                        if vref.is_none_or(|v| v[i]) {
435                            *x = x.floor();
436                        }
437                    }
438                }
439                ScalarOp::Ceil => {
440                    for (i, x) in data.iter_mut().enumerate() {
441                        if vref.is_none_or(|v| v[i]) {
442                            *x = x.ceil();
443                        }
444                    }
445                }
446                ScalarOp::Neg => {
447                    for (i, x) in data.iter_mut().enumerate() {
448                        if vref.is_none_or(|v| v[i]) {
449                            *x = -(*x);
450                        }
451                    }
452                }
453            }
454            Ok(())
455        } else {
456            Err(format!(
457                "col_scalar_op: column '{col}' is not Float64 (type: {})",
458                self.columns[idx].dtype_name()
459            ))
460        }
461    }

480    /// Compute element-wise binary operation between two numeric columns and write result column.
481    pub fn col_binary_op(
482        &mut self,
483        out_col: &str,
484        left_col: &str,
485        op: BinaryOp,
486        right_col: &str,
487    ) -> Result<(), String> {
488        let li = self.resolve(left_col)?;
489        let ri = self.resolve(right_col)?;
490        let nrows = self.nrows();
491        let left_data: Vec<f64> = match &self.columns[li] {
492            ColumnStore::Float64 { data, .. } => data.clone(),
493            ColumnStore::Int64 { data, .. } => data.iter().map(|v| *v as f64).collect(),
494            t => {
495                return Err(format!(
496                    "col_binary_op: left column '{left_col}' type '{}' is not numeric",
497                    t.dtype_name()
498                ))
499            }
500        };
501        let right_data: Vec<f64> = match &self.columns[ri] {
502            ColumnStore::Float64 { data, .. } => data.clone(),
503            ColumnStore::Int64 { data, .. } => data.iter().map(|v| *v as f64).collect(),
504            t => {
505                return Err(format!(
506                    "col_binary_op: right column '{right_col}' type '{}' is not numeric",
507                    t.dtype_name()
508                ))
509            }
510        };
511        let left_valid = match &self.columns[li] {
512            ColumnStore::Float64 { validity, .. } | ColumnStore::Int64 { validity, .. } => {
513                validity.clone()
514            }
515            _ => None,
516        };
517        let right_valid = match &self.columns[ri] {
518            ColumnStore::Float64 { validity, .. } | ColumnStore::Int64 { validity, .. } => {
519                validity.clone()
520            }
521            _ => None,
522        };
523        let mut out_data = Vec::with_capacity(nrows);
524        let mut out_validity: Option<Vec<bool>> = None;
525        for i in 0..nrows {
526            let l_ok = left_valid.as_ref().is_none_or(|v| v[i]);
527            let r_ok = right_valid.as_ref().is_none_or(|v| v[i]);
528            if !l_ok || !r_ok {
529                out_data.push(0.0f64);
530                out_validity.get_or_insert_with(|| vec![true; nrows])[i] = false;
531            } else {
532                let l = left_data[i];
533                let r = right_data[i];
534                let result = match op {
535                    BinaryOp::Add => l + r,
536                    BinaryOp::Sub => l - r,
537                    BinaryOp::Mul => l * r,
538                    BinaryOp::Div => {
539                        if r == 0.0 {
540                            out_data.push(f64::NAN);
541                            if out_validity.is_none() {
542                                out_validity = Some(vec![true; nrows]);
543                            }
544                            out_validity.as_mut().unwrap()[i] = false;
545                            continue;
546                        }
547                        l / r
548                    }
549                    BinaryOp::Min => l.min(r),
550                    BinaryOp::Max => l.max(r),
551                };
552                out_data.push(result);
553            }
554        }
555        let store = ColumnStore::Float64 {
556            data: out_data,
557            validity: out_validity,
558        };
559        if let Some(oi) = self.column_names.iter().position(|n| n == out_col) {
560            self.columns[oi] = store;
561        } else {
562            self.column_names.push(out_col.to_string());
563            self.columns.push(store);
564        }
565        Ok(())
566    }
567    /// Reduce numeric column to single value using selected aggregation.
568    pub fn col_reduce(&self, col: &str, op: ReduceOp) -> Result<Option<f64>, String> {
569        let idx = self.resolve(col)?;
570        let vals = match &self.columns[idx] {
571            ColumnStore::Float64 { .. } => self.columns[idx]
572                .valid_f64s()
573                .ok_or_else(|| "impossible".to_string())?,
574            ColumnStore::Int64 { data, validity } => {
575                let base: Vec<f64> = data.iter().map(|v| *v as f64).collect();
576                if let Some(v) = validity {
577                    base.iter()
578                        .zip(v.iter())
579                        .filter_map(|(x, ok)| if *ok { Some(*x) } else { None })
580                        .collect()
581                } else {
582                    base
583                }
584            }
585            t => {
586                return Err(format!(
587                    "col_reduce: column '{col}' type '{}' is not numeric",
588                    t.dtype_name()
589                ))
590            }
591        };
592        if vals.is_empty() {
593            return Ok(None);
594        }
595        let result = match op {
596            ReduceOp::Count => Some(vals.len() as f64),
597            ReduceOp::Sum => Some(vals.iter().sum()),
598            ReduceOp::Mean => Some(vals.iter().sum::<f64>() / vals.len() as f64),
599            ReduceOp::Min => vals.iter().cloned().reduce(f64::min),
600            ReduceOp::Max => vals.iter().cloned().reduce(f64::max),
601            ReduceOp::Var => {
602                let mean = vals.iter().sum::<f64>() / vals.len() as f64;
603                let var = vals.iter().map(|x| (x - mean).powi(2)).sum::<f64>() / vals.len() as f64;
604                Some(var)
605            }
606            ReduceOp::Std => {
607                let mean = vals.iter().sum::<f64>() / vals.len() as f64;
608                let var = vals.iter().map(|x| (x - mean).powi(2)).sum::<f64>() / vals.len() as f64;
609                Some(var.sqrt())
610            }
611        };
612        Ok(result)
613    }
614    /// Build boolean mask by comparing numeric column against scalar value.
615    pub fn filter_mask(&self, col: &str, op: CmpOp, val: f64) -> Result<Vec<bool>, String> {
616        let idx = self.resolve(col)?;
617        let nrows = self.nrows();
618        let mut mask = vec![false; nrows];
619        match &self.columns[idx] {
620            ColumnStore::Float64 { data, validity } => {
621                for (i, x) in data.iter().enumerate() {
622                    if validity.as_ref().is_none_or(|v| v[i]) {
623                        mask[i] = cmp_f64(*x, op, val);
624                    }
625                }
626            }
627            ColumnStore::Int64 { data, validity } => {
628                for (i, x) in data.iter().enumerate() {
629                    if validity.as_ref().is_none_or(|v| v[i]) {
630                        mask[i] = cmp_f64(*x as f64, op, val);
631                    }
632                }
633            }
634            t => {
635                return Err(format!(
636                    "filter_mask: column '{col}' type '{}' is not numeric",
637                    t.dtype_name()
638                ))
639            }
640        }
641        Ok(mask)
642    }
643    /// Filter all columns by boolean mask and return new VecFrame.
644    pub fn apply_mask(&self, mask: &[bool]) -> Result<VecFrame, String> {
645        if mask.len() != self.nrows() {
646            return Err(format!(
647                "apply_mask: mask length {} != nrows {}",
648                mask.len(),
649                self.nrows()
650            ));
651        }
652        let columns: Vec<ColumnStore> = self.columns.iter().map(|c| c.filter(mask)).collect();
653        Ok(VecFrame {
654            column_names: self.column_names.clone(),
655            columns,
656        })
657    }
658    /// Cast named column to target type in place.
659    pub fn col_cast(&mut self, col: &str, dtype: &str) -> Result<(), String> {
660        let idx = self.resolve(col)?;
661        let nrows = self.nrows();
662        let new_store = match (self.columns[idx].dtype_name(), dtype) {
663            ("float64", "int64") => {
664                if let ColumnStore::Float64 { data, validity } = &self.columns[idx] {
665                    ColumnStore::Int64 {
666                        data: data.iter().map(|v| *v as i64).collect(),
667                        validity: validity.clone(),
668                    }
669                } else {
670                    unreachable!()
671                }
672            }
673            ("int64", "float64") => {
674                if let ColumnStore::Int64 { data, validity } = &self.columns[idx] {
675                    ColumnStore::Float64 {
676                        data: data.iter().map(|v| *v as f64).collect(),
677                        validity: validity.clone(),
678                    }
679                } else {
680                    unreachable!()
681                }
682            }
683            ("float64", "text") => {
684                if let ColumnStore::Float64 { data, validity } = &self.columns[idx] {
685                    ColumnStore::Text {
686                        data: data.iter().map(|v| format!("{v}")).collect(),
687                        validity: validity.clone(),
688                    }
689                } else {
690                    unreachable!()
691                }
692            }
693            ("int64", "text") => {
694                if let ColumnStore::Int64 { data, validity } = &self.columns[idx] {
695                    ColumnStore::Text {
696                        data: data.iter().map(|v| format!("{v}")).collect(),
697                        validity: validity.clone(),
698                    }
699                } else {
700                    unreachable!()
701                }
702            }
703            (from, to) if from == to => return Ok(()),
704            (from, to) => {
705                let _ = nrows;
706                return Err(format!("col_cast: cannot cast '{col}' from {from} to {to}"));
707            }
708        };
709        self.columns[idx] = new_store;
710        Ok(())
711    }
712    /// Reduce multiple columns in parallel and return name-to-result map.
713    pub fn par_reduce(&self, cols: &[&str], op: ReduceOp) -> HashMap<String, Option<f64>> {
714        cols.par_iter()
715            .map(|col| {
716                let result = self.col_reduce(col, op).ok().flatten();
717                (col.to_string(), result)
718            })
719            .collect()
720    }
721    /// Apply scalar operation across multiple Float64 columns in parallel.
722    pub fn par_scalar_op(&mut self, cols: &[&str], op: ScalarOp, val: f64) -> Result<(), String> {
723        let indices: Result<Vec<usize>, String> = cols.iter().map(|c| self.resolve(c)).collect();
724        let indices = indices?;
725        let updates: Vec<(usize, Vec<f64>)> = indices
726            .par_iter()
727            .filter_map(|&idx| {
728                if let ColumnStore::Float64 { data, validity } = &self.columns[idx] {
729                    let vref = validity.as_ref();
730                    let new_data: Vec<f64> = data
731                        .iter()
732                        .enumerate()
733                        .map(|(i, &x)| {
734                            if vref.is_none_or(|v| v[i]) {
735                                apply_scalar_f64(x, op, val)
736                            } else {
737                                x
738                            }
739                        })
740                        .collect();
741                    Some((idx, new_data))
742                } else {
743                    None
744                }
745            })
746            .collect();
747        for (idx, new_data) in updates {
748            if let ColumnStore::Float64 { data, .. } = &mut self.columns[idx] {
749                *data = new_data;
750            }
751        }
752        Ok(())
753    }
754}

762impl DataFrame {
763    /// Parallel element-wise transform on a column.
764    /// Uses rayon when the column has 10,000+ rows; sequential otherwise.
765    pub fn par_apply_column<F>(&mut self, col: ColRef, f: F) -> Result<(), String>
766    where
767        F: Fn(&CellValue) -> CellValue + Send + Sync,
768    {
769        let ci = self.resolve_col(col)?;
770        let col_data = &self.data[ci];
771        const PAR_THRESHOLD: usize = 10_000;
772
773        let new_col: Vec<CellValue> = if col_data.len() >= PAR_THRESHOLD {
774            col_data.par_iter().map(&f).collect()
775        } else {
776            col_data.iter().map(f).collect()
777        };
778        self.data[ci] = new_col;
779        Ok(())
780    }
781}
782/// Infer best column storage type from a vector of cell values.
783fn infer_column(cells: &[CellValue]) -> ColumnStore {
784    let mut has_num = false;
785    let mut has_bool = false;
786    let mut has_text = false;
787    let mut has_nil = false;
788    for c in cells.iter() {
789        match c {
790            CellValue::Number(_) => has_num = true,
791            CellValue::Bool(_) => has_bool = true,
792            CellValue::Text(_) => has_text = true,
793            CellValue::Nil => has_nil = true,
794        }
795    }
796    let n = cells.len();
797    if has_bool && !has_num && !has_text {
798        let mut data = Vec::with_capacity(n);
799        let mut validity = if has_nil { Some(vec![true; n]) } else { None };
800        for (i, c) in cells.iter().enumerate() {
801            match c {
802                CellValue::Bool(b) => data.push(*b),
803                CellValue::Nil => {
804                    data.push(false);
805                    if let Some(v) = &mut validity {
806                        v[i] = false;
807                    }
808                }
809                _ => data.push(false),
810            }
811        }
812        ColumnStore::Bool { data, validity }
813    } else if has_text && !has_num && !has_bool {
814        let mut data = Vec::with_capacity(n);
815        let mut validity = if has_nil { Some(vec![true; n]) } else { None };
816        for (i, c) in cells.iter().enumerate() {
817            match c {
818                CellValue::Text(s) => data.push(s.clone()),
819                CellValue::Nil => {
820                    data.push(String::new());
821                    if let Some(v) = &mut validity {
822                        v[i] = false;
823                    }
824                }
825                _ => data.push(String::new()),
826            }
827        }
828        ColumnStore::Text { data, validity }
829    } else {
830        let mut data = Vec::with_capacity(n);
831        let mut validity = if has_nil || has_text || has_bool {
832            Some(vec![true; n])
833        } else {
834            None
835        };
836        for (i, c) in cells.iter().enumerate() {
837            match c {
838                CellValue::Number(v) => data.push(*v),
839                CellValue::Nil => {
840                    data.push(0.0);
841                    if let Some(v) = &mut validity {
842                        v[i] = false;
843                    }
844                }
845                CellValue::Bool(b) => data.push(if *b { 1.0 } else { 0.0 }),
846                CellValue::Text(_) => {
847                    data.push(0.0);
848                    if let Some(v) = &mut validity {
849                        v[i] = false;
850                    }
851                }
852            }
853        }
854        ColumnStore::Float64 { data, validity }
855    }
856}
857/// Apply scalar operation to one f64 value and return transformed value.
858fn apply_scalar_f64(x: f64, op: ScalarOp, val: f64) -> f64 {
859    match op {
860        ScalarOp::Add => x + val,
861        ScalarOp::Sub => x - val,
862        ScalarOp::Mul => x * val,
863        ScalarOp::Div => x / val,
864        ScalarOp::Abs => x.abs(),
865        ScalarOp::Sqrt => x.sqrt(),
866        ScalarOp::Floor => x.floor(),
867        ScalarOp::Ceil => x.ceil(),
868        ScalarOp::Neg => -x,
869    }
870}
871/// Compare one value against scalar and return predicate result.
872fn cmp_f64(x: f64, op: CmpOp, val: f64) -> bool {
873    match op {
874        CmpOp::Lt => x < val,
875        CmpOp::Le => x <= val,
876        CmpOp::Gt => x > val,
877        CmpOp::Ge => x >= val,
878        CmpOp::Eq => (x - val).abs() < f64::EPSILON,
879        CmpOp::Ne => (x - val).abs() >= f64::EPSILON,
880    }

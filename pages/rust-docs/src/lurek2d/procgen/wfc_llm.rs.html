1//! LLM-assisted helper layer for turning natural-language intent into concrete WFC tiles, weights, and adjacency rules.
2//! The file handles prompt shaping and response parsing so language-model output can become structured generator input rather than loose text.
3//! Keeping that translation here isolates the experimental boundary between authored prompts and deterministic procedural systems.
4//! Functionally this file delivers an assisted authoring path for bootstrapping WFC constraints from descriptive input.
5
6use crate::procgen::{WfcOpts, WfcRules, WfcTile};
7use std::collections::HashMap;
8
9/// Parse a `serde_json::Value` LLM response into [`WfcOpts`].
10///
11/// Expected JSON format:
12/// `{"tiles":[{"id":0,"weight":1.0},...], "adjacencies":{"0":[1,2,3],...}}`
13///
14/// Returns `None` if the tile list is absent or empty.
15pub fn parse_llm_wfc_response(
16    val: &serde_json::Value,
17    width: u32,
18    height: u32,
19    seed: u64,
20    max_attempts: u32,
21) -> Option<WfcOpts> {
22    let tiles_arr = val.get("tiles")?.as_array()?;
23    let mut tiles: Vec<WfcTile> = Vec::new();
24    for t in tiles_arr {
25        let id = t.get("id")?.as_u64()? as u32;
26        let weight = t.get("weight").and_then(|v| v.as_f64()).unwrap_or(1.0) as f32;
27        tiles.push(WfcTile { id, weight });
28    }
29    if tiles.is_empty() {
30        return None;
31    }
32
33    let adjacencies = parse_adjacency_object(val.get("adjacencies"));
34
35    Some(WfcOpts {
36        width,
37        height,
38        tiles,
39        rules: WfcRules { adjacencies },
40        seed,
41        max_attempts,
42    })
43}

45/// Parse only the adjacency rules from an LLM response.
46///
47/// Expected JSON: `{"adjacencies":{"0":[1,2],...}}`
48///
49/// Returns an empty map when the field is absent or malformed.
50pub fn parse_llm_constraints(val: &serde_json::Value) -> HashMap<u32, Vec<u32>> {
51    parse_adjacency_object(val.get("adjacencies"))
52}
53
54fn parse_adjacency_object(obj_val: Option<&serde_json::Value>) -> HashMap<u32, Vec<u32>> {
55    let mut out = HashMap::new();
56    let Some(obj) = obj_val.and_then(|v| v.as_object()) else {
57        return out;
58    };
59    for (k, v) in obj {
60        let Ok(id) = k.parse::<u32>() else { continue };
61        let Some(arr) = v.as_array() else { continue };
62        let neighbors: Vec<u32> = arr
63            .iter()
64            .filter_map(|n| n.as_u64().map(|x| x as u32))
65            .collect();
66        out.insert(id, neighbors);
67    }
68    out
69}

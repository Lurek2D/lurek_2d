1//! Lightweight projection layer for sampled noise grids that need storage, cell access, and quick grayscale export without depending on higher rendering systems.
2//! The file keeps tileable Perlin-backed values in a compact grid form and exposes them in a way that is useful for previews, tooling, and texture-oriented workflows.
3//! Functionally this file delivers a small bridge from procedural scalar fields to inspectable pixel-ready grid data.
4
5use crate::procgen::noise::perlin_noise_periodic;
6use crate::procgen::scalar_map_to_rgba_bytes;
7
8/// Flat noise grid with tiling-Perlin cell values in 0.0–1.0.
9pub struct NoiseGrid {
10    /// Grid width in cells.
11    pub width: u32,
12    /// Grid height in cells.
13    pub height: u32,
14    /// Row-major cell values; index = `y * width + x`.
15    pub cells: Vec<f32>,
16}
17
18impl NoiseGrid {
19    /// Build a tileable Perlin noise grid at the given `scale`; scale is clamped to >= 1e-6.
20    pub fn from_perlin(width: u32, height: u32, scale: f64) -> Self {
21        let scale = scale.max(1e-6);
22        let px = width as f64 * scale;
23        let py = height as f64 * scale;
24        let size = (width * height) as usize;
25        let mut cells = Vec::with_capacity(size);
26        for y in 0..height {
27            for x in 0..width {
28                let nx = x as f64 * scale;
29                let ny = y as f64 * scale;
30                let v = (perlin_noise_periodic(nx, ny, px, py) * 0.5 + 0.5).clamp(0.0, 1.0) as f32;
31                cells.push(v);
32            }
33        }
34        Self {
35            width,
36            height,
37            cells,
38        }
39    }
40    /// Convert the cell grid to a flat grayscale RGBA byte buffer at 4 bytes per cell.
41    pub fn to_rgba_bytes(&self) -> Vec<u8> {
42        scalar_map_to_rgba_bytes(&self.cells)
43    }
44}

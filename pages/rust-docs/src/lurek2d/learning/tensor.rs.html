1//! Defines lightweight tensor containers and helpers used by learning components.
2//! Stores shape metadata and flat row-major data for predictable indexing behavior.
3//! Provides indexing, flattening, and conversion utilities needed by model layers.
4//! Includes compact numeric operations that support CPU learning pipelines.
5
6use ndarray::ArrayD;
7use tract_onnx::prelude::{IntoTensor, Tensor};
8
9/// Flat f32 tensor with explicit row-major shape metadata.
10#[derive(Debug, Clone)]
11pub struct LurekTensor {
12    /// Dimension sizes in row-major order.
13    pub shape: Vec<usize>,
14    /// Flat element data in row-major order.
15    pub data: Vec<f32>,
16}
17
18impl LurekTensor {
19    /// Create a tensor with the given `shape` and flat `data`.
20    pub fn new(shape: Vec<usize>, data: Vec<f32>) -> Self {
21        Self { shape, data }
22    }
23
24    /// Create a zero-filled tensor for the given `shape`.
25    pub fn zeros(shape: Vec<usize>) -> Self {
26        let len = shape.iter().product();
27        Self {
28            shape,
29            data: vec![0.0; len],
30        }
31    }
32
33    /// Returns the total number of tensor elements.
34    pub fn len(&self) -> usize {
35        self.data.len()
36    }
37
38    /// True when there are no elements.
39    pub fn is_empty(&self) -> bool {
40        self.data.is_empty()
41    }
42
43    /// Return the element at the given multi-dimensional indices (zero-based, row-major).
44    pub fn get_element(&self, indices: &[usize]) -> Option<f32> {
45        self.data.get(self.flat_index(indices)?).copied()
46    }
47
48    /// Convert multi-dimensional zero-based indices to a flat row-major offset.
49    pub fn flat_index(&self, indices: &[usize]) -> Option<usize> {
50        if indices.len() != self.shape.len() {
51            return None;
52        }
53        let mut idx = 0usize;
54        let mut stride = 1usize;
55        for (&i, &s) in indices.iter().zip(self.shape.iter()).rev() {
56            if i >= s {
57                return None;
58            }
59            idx += i * stride;
60            stride *= s;
61        }
62        Some(idx)
63    }
64
65    /// Return a flattened copy with shape `[len]`.
66    pub fn flatten(&self) -> Self {
67        Self {
68            shape: vec![self.data.len()],
69            data: self.data.clone(),
70        }
71    }
72
73    /// Build a tract `Tensor` from this handle for use as a model input.
74    pub fn to_tract_tensor(&self) -> Result<Tensor, String> {
75        let arr = ArrayD::<f32>::from_shape_vec(self.shape.clone(), self.data.clone())
76            .map_err(|e| e.to_string())?;
77        Ok(arr.into_tensor())
78    }
79}

81/// Multiply matrix A `[m x k]` by matrix B `[k x n]` and optionally add a bias `[n]`.
82pub fn gemm(m: usize, k: usize, n: usize, a: &[f32], b: &[f32], bias: Option<&[f32]>) -> Vec<f32> {
83    let mut out = vec![0.0; m * n];
84    for i in 0..m {
85        for j in 0..n {
86            let mut sum = bias.map_or(0.0, |v| v[j]);
87            for p in 0..k {
88                sum += a[i * k + p] * b[p * n + j];
89            }
90            out[i * n + j] = sum;
91        }
92    }
93    out

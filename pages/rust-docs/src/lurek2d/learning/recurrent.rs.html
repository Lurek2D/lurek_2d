1//! Provides recurrent sequence-learning layers including LSTM and GRU style stateful blocks.
2//! Stores gate parameters in flat row-major buffers suitable for CPU forward evaluation.
3//! Executes timestep iteration while carrying hidden-state context across sequence positions.
4//! Integrates with evolutionary parameter interfaces for genome-based optimization workflows.
5//! Offers compact recurrent primitives for temporal modeling without heavyweight dependencies.
6//! Serves as a reusable foundation for sequence tasks in higher-level learning engines.
7
8use crate::learning::EvolutionaryLayer;
9
10/// CPU LSTM layer with combined gate matrices.
11pub struct LstmLayer {
12    pub input_size: usize,
13    pub hidden_size: usize,
14    /// `[4H, D]` flattened row-major.
15    pub w_gate: Vec<f32>,
16    /// `[4H, H]` flattened row-major.
17    pub u_gate: Vec<f32>,
18    /// `[4H]`.
19    pub b_gate: Vec<f32>,
20}
21
22impl LstmLayer {
23    /// Create a zero-initialized LSTM layer.
24    pub fn new(input_size: usize, hidden_size: usize) -> Self {
25        Self {
26            input_size,
27            hidden_size,
28            w_gate: vec![0.0; 4 * hidden_size * input_size],
29            u_gate: vec![0.0; 4 * hidden_size * hidden_size],
30            b_gate: vec![0.0; 4 * hidden_size],
31        }
32    }
33
34    /// Execute one recurrent step.
35    pub fn step(&self, x: &[f32], prev_h: &[f32], prev_c: &[f32]) -> Result<(Vec<f32>, Vec<f32>), String> {
36        let h = self.hidden_size;
37        if x.len() != self.input_size {
38            return Err("LstmLayer::step input size mismatch".to_string());
39        }
40        if prev_h.len() != h || prev_c.len() != h {
41            return Err("LstmLayer::step state size mismatch".to_string());
42        }
43
44        let mut gates = vec![0.0f32; 4 * h];
45        for (g, gate_slot) in gates.iter_mut().enumerate().take(4 * h) {
46            let mut sum = self.b_gate[g];
47            for (i, &xv) in x.iter().enumerate().take(self.input_size) {
48                sum += self.w_gate[g * self.input_size + i] * xv;
49            }
50            for (j, &hv) in prev_h.iter().enumerate().take(h) {
51                sum += self.u_gate[g * h + j] * hv;
52            }
53            *gate_slot = sum;
54        }
55
56        let mut next_h = vec![0.0f32; h];
57        let mut next_c = vec![0.0f32; h];
58        for i in 0..h {
59            let in_gate = sigmoid(gates[i]);
60            let forget_gate = sigmoid(gates[h + i]);
61            let cell_candidate = gates[2 * h + i].tanh();
62            let out_gate = sigmoid(gates[3 * h + i]);
63
64            next_c[i] = forget_gate * prev_c[i] + in_gate * cell_candidate;
65            next_h[i] = out_gate * next_c[i].tanh();
66        }
67
68        Ok((next_h, next_c))
69    }
70}
71
72impl EvolutionaryLayer for LstmLayer {
73    fn param_count(&self) -> usize {
74        4 * self.hidden_size * (self.input_size + self.hidden_size + 1)
75    }
76
77    fn set_weights(&mut self, weights: &[f32]) -> bool {
78        let expected = self.param_count();
79        if weights.len() != expected {
80            return false;
81        }
82
83        let h = self.hidden_size;
84        let d = self.input_size;
85        let w_offset = 4 * h * d;
86        let u_offset = w_offset + 4 * h * h;
87
88        self.w_gate.copy_from_slice(&weights[..w_offset]);
89        self.u_gate.copy_from_slice(&weights[w_offset..u_offset]);
90        self.b_gate.copy_from_slice(&weights[u_offset..]);
91        true
92    }
93
94    fn get_weights(&self) -> Vec<f32> {
95        let mut out = Vec::with_capacity(self.param_count());
96        out.extend_from_slice(&self.w_gate);
97        out.extend_from_slice(&self.u_gate);
98        out.extend_from_slice(&self.b_gate);
99        out
100    }
101}

103/// CPU GRU layer with combined gate matrices.
104pub struct GruLayer {
105    pub input_size: usize,
106    pub hidden_size: usize,
107    /// `[3H, D]` flattened row-major.
108    pub w_gate: Vec<f32>,
109    /// `[3H, H]` flattened row-major.
110    pub u_gate: Vec<f32>,
111    /// `[3H]`.
112    pub b_gate: Vec<f32>,
113}
114
115impl GruLayer {
116    /// Create a zero-initialized GRU layer.
117    pub fn new(input_size: usize, hidden_size: usize) -> Self {
118        Self {
119            input_size,
120            hidden_size,
121            w_gate: vec![0.0; 3 * hidden_size * input_size],
122            u_gate: vec![0.0; 3 * hidden_size * hidden_size],
123            b_gate: vec![0.0; 3 * hidden_size],
124        }
125    }
126
127    /// Execute one recurrent step.
128    pub fn step(&self, x: &[f32], prev_h: &[f32]) -> Result<Vec<f32>, String> {
129        let h = self.hidden_size;
130        if x.len() != self.input_size {
131            return Err("GruLayer::step input size mismatch".to_string());
132        }
133        if prev_h.len() != h {
134            return Err("GruLayer::step state size mismatch".to_string());
135        }
136
137        let mut z = vec![0.0f32; h];
138        let mut r = vec![0.0f32; h];
139        for i in 0..h {
140            let mut z_sum = self.b_gate[i];
141            let mut r_sum = self.b_gate[h + i];
142            for (j, &xv) in x.iter().enumerate().take(self.input_size) {
143                z_sum += self.w_gate[i * self.input_size + j] * xv;
144                r_sum += self.w_gate[(h + i) * self.input_size + j] * xv;
145            }
146            for (j, &hv) in prev_h.iter().enumerate().take(h) {
147                z_sum += self.u_gate[i * h + j] * hv;
148                r_sum += self.u_gate[(h + i) * h + j] * hv;
149            }
150            z[i] = sigmoid(z_sum);
151            r[i] = sigmoid(r_sum);
152        }
153
154        let mut h_tilde = vec![0.0f32; h];
155        for (i, h_tilde_slot) in h_tilde.iter_mut().enumerate().take(h) {
156            let mut sum = self.b_gate[2 * h + i];
157            for (j, &xv) in x.iter().enumerate().take(self.input_size) {
158                sum += self.w_gate[(2 * h + i) * self.input_size + j] * xv;
159            }
160            for (j, &hv) in prev_h.iter().enumerate().take(h) {
161                sum += self.u_gate[(2 * h + i) * h + j] * (r[j] * hv);
162            }
163            *h_tilde_slot = sum.tanh();
164        }
165
166        let mut next_h = vec![0.0f32; h];
167        for i in 0..h {
168            next_h[i] = (1.0 - z[i]) * prev_h[i] + z[i] * h_tilde[i];
169        }
170        Ok(next_h)
171    }
172}

205fn sigmoid(x: f32) -> f32 {
206    1.0 / (1.0 + (-x).exp())

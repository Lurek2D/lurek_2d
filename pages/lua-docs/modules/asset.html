Module

lurek.asset

`lurek.asset` — asset registry and ref-counted media cache bindings. Registers `lurek.asset.*` into the global `lurek` table via `register()`. Bridges 23 Lua-callable functions and 2 `LAssetHandle` methods via `mlua`. Supported asset types: `image`, `font`, `audio`, `music`, `text`, `toml`, `json`, `obj`, `shader`, `lua`. Text-like types have their file content read and cached; binary types store a path reference only. Assets can be annotated with a display name, a group label, and a tag set. Query functions (`findByName`, `findByGroup`, `findByTag`, `findByType`) return arrays of `LAssetHandle` values that can be iterated by scripts. See `docs/specs/asset.md` for the full API specification.

load

lurek.asset.load(path: string, asset_type: string, opts: table?) -> LAssetHandle

Loads and caches an asset by path and type, returning a ref-counted handle.

functionsrc/lua_api/asset_api.rs:212
Example · Exact example · asset.lua
do
    -- Minimal load: text type reads file content immediately.
    local h = lurek.asset.load(PATH_JSON, "text")
    print("loaded: " .. h:type())
    lurek.asset.unload(h)

    -- load type=toml
    local ht = lurek.asset.load(PATH_TOML, "toml")
    print("toml loaded: " .. tostring(lurek.asset.isLoaded(ht)))
    lurek.asset.unload(ht)

    -- load type=json
    local hj = lurek.asset.load(PATH_JSON, "json")
    print("json loaded: " .. tostring(lurek.asset.isLoaded(hj)))
    lurek.asset.unload(hj)

    -- load type=lua
    local hl = lurek.asset.load(PATH_LUA, "lua")
    print("lua loaded: " .. tostring(lurek.asset.isLoaded(hl)))
    lurek.asset.unload(hl)

    -- load type=shader
    local hs = lurek.asset.load(PATH_SHADER, "shader")
    print("shader loaded: " .. tostring(lurek.asset.isLoaded(hs)))
    lurek.asset.unload(hs)

    -- load type=obj (any text file works for raw OBJ geometry)
    local ho = lurek.asset.load(PATH_LUA, "obj")
    print("obj loaded: " .. tostring(lurek.asset.isLoaded(ho)))
    lurek.asset.unload(ho)

    -- load type=music (binary path reference only; no file content cached)
    local hm = lurek.asset.load(PATH_BIN, "music")
    print("music loaded: " .. tostring(lurek.asset.isLoaded(hm)))
    lurek.asset.unload(hm)

    -- load type=audio (same as music but semantically a sound effect)
    local ha = lurek.asset.load(PATH_BIN, "audio")
    print("audio loaded: " .. tostring(lurek.asset.isLoaded(ha)))
    lurek.asset.unload(ha)
end

preload

lurek.asset.preload(paths: table, callback: any) -> nil

Synchronously loads a batch of assets and fires `callback(loaded, total)` after each item.

functionsrc/lua_api/asset_api.rs:272
Example · Exact example · asset.lua
do
    local results = {}
    lurek.asset.preload(
        {
            {PATH_JSON,   "json"},
            {PATH_TOML,   "toml"},
            {PATH_LUA,    "lua"},
        },
        function(loaded, total)
            if loaded ~= nil then
                table.insert(results, loaded .. "/" .. tostring(total))
            else
                print("preload done: " .. table.concat(results, ", "))
            end
        end
    )
    lurek.asset.clear()
end

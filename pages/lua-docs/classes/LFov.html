LFov

Lua-side wrapper for a tile-grid recursive-shadowcasting FOV.

Docs home/lurek.visibility
Class Example · Exact example · visibility.lua
do
    local fov = lurek.visibility.newFov({ width = 20, height = 20, range = 8 })
    fov:setBlocker(function(x, y)
        return x == 10 and y >= 6 and y <= 14
    end)
    fov:compute(5, 10)
    print("LFov:setBlocker visible_12_10=" .. tostring(fov:isVisible(12, 10)))
end

eachVisible

    fov:compute(10, 10)

        count = count + 1
